"""D1 driver core with deterministic trusted fake ports; RED-first, no physical effect.

Contract for the FUTURE ``deploy/oracle9_w1_driver.py`` (DRIVER-DESIGN v3 plus

Minimal stable API of the future module (``runpy`` namespace):

- Constants: ``PURE_PATH`` "/opt/maskgw/lab/oracle9_w1_controller.py",
  ``PURE_SHA`` (4c155d76...), ``PURE_BYTES`` 22453, ``GUARD_AS_BYTES`` 384 MiB,

- ``load_pure(reader) -> types.ModuleType``: calls
  ``reader.read(PURE_PATH, PURE_BYTES + 1)`` once, requires exact ``bytes`` of
  size PURE_BYTES and PURE_SHA, then executes THOSE bytes as a private module
  (``__file__ == PURE_PATH``) registered in ``sys.modules``. Any other input:
  ``DriverRefusal("LOAD")`` BEFORE execution. No path/hash/size parameter.
- ``Ports(reader, guard, clock, link, ids, factory, pair, threads)`` and

def exact_int(value: object, low: int, high: int) -> bool:
    return type(value) is int and low <= value <= high


def optional(value: object, check: Callable[[object], bool]) -> bool:
    return value is None or check(value)


def check_result(event: dict[str, Any]) -> bool:
    return (
        exact_int(event["case"], 1, 24)
        and exact_int(event["next_seq"], 2, 49)
        and event["verdict"] in ("PASS", "FAIL")
        and event["kind"] in ("RESULT", "ERROR", "FATAL", "PENDING")
        and optional(event["rows"], lambda v: exact_int(v, 0, 1000))
        and optional(event["truncated"], lambda v: type(v) is bool)
        and optional(event["bytes"], lambda v: exact_int(v, 0, SINK_CAP + 1))
        and optional(event["sink"], lambda v: v in SINKS)
        and optional(event["reason"], lambda v: v in RESULT_REASONS)
        and optional(event["release"], lambda v: v in ("RELEASED", "ALREADY_RELEASED", "HELD"))
        and optional(event["stop_applied"], lambda v: type(v) is bool)
    )


def check_armed(event: dict[str, Any]) -> bool:
    if event["case"] == T4 and type(event["case"]) is int:
        return event["next_seq"] == 46 and event["live_window_ms"] is None
    return (
        event["case"] == T6
        and type(event["case"]) is int
        and type(event["next_seq"]) is int
        and event["next_seq"] == 51
        and exact_int(event["live_window_ms"], 1000, 30000)
    )


def check_summary(event: dict[str, Any]) -> bool:
    failed = event["failed"]
    return (
        type(failed) is list
        and all(exact_int(case, 1, 24) for case in failed)
        and failed == sorted(set(failed))
        and T5 in failed
        and event["global_pass"] is False
        and type(event["prefix_cases"]) is int
        and event["prefix_cases"] == 24
    )

def events_of(lines: list[bytes]) -> list[dict[str, Any]]:
    assert len(lines) <= 56
    result: list[dict[str, Any]] = []
    for line in lines:
        assert type(line) is bytes
        assert 0 < len(line) <= 512
        assert b"\n" not in line
        for leak in LEAKS:
            assert leak.encode() not in line
        event = json.loads(line)
        assert canonical(event) == line
        assert set(event) == KEYS[event["e"]]
        assert CHECKS[event["e"]](event)
        result.append(event)
    return result

class Clock:
    def __init__(self) -> None:
        self.now = START
        self.sleeps: list[float] = []

    def monotonic(self) -> float:
        return self.now

    def sleep_until(self, when: float) -> None:
        self.sleeps.append(when)
        self.now = max(self.now, when)

class Link:
    def __init__(self, world: World) -> None:
        self.world = world
        self.lines: list[bytes] = []
        self.pending: list[tuple[bytes, float]] = []
        self.hold = False
        self.last: tuple[str, int] = ("", 0)
        self.reads: list[tuple[tuple[str, int], float, float]] = []

    def write(self, line: bytes) -> bool:
        self.world.unlocked("write")
        self.lines.append(line)
        event = json.loads(line)
        self.last = (event["e"], event.get("case", 0))
        self.world.log.append(("write", event["e"], event.get("case", 0)))
        if event["e"] == "ARMED" and event["case"] == T6 and self.world.ledger is not None:
            self.world.armed_status = self.world.ledger.status
        self.react(event)
        return True

    def react(self, event: dict[str, Any]) -> None:
        kind, case = event["e"], event.get("case", 0)
        seq = event.get("next_seq", 0)
        script = self.world.script
        if kind == "READY":
            self.queue(GO, case, seq, script.get("go_delay", {}).get(case, 0.0))
        elif kind == "RESULT":
            self.queue(ACK, case, seq, script.get("ack_delay", {}).get(case, 0.0))
        elif kind == "ARMED" and case == T4:
            self.queue(script.get("t4", STOP), T4, seq, 0.0)
        elif kind == "ARMED" and case == T6:
            if script.get("after_armed"):
                self.pending.append((frame(ACK, T6, 50), 0.0))
            else:
                self.hold = True

    def queue(self, kind: int, case: int, seq: int, delay: float) -> None:
        if (kind, case) in self.world.script.get("eof", ()):
            return
        raw = self.world.script.get("frames", {}).get((kind, case), frame(kind, case, seq))
        self.pending.append((raw, delay))

    def read_init(self, *, deadline: float) -> bytes | None:
        self.world.unlocked("read_init")
        self.world.log.append(("read_init", deadline))
        result: bytes | None = self.world.script.get("init", PACKET)
        return result

    def read_control(self, *, deadline: float) -> bytes | None:
        self.world.unlocked("read_control")
        clock = self.world.clock
        self.reads.append((self.last, deadline, clock.now))
        self.world.log.append(("read_control", self.last))
        if not self.pending:
            if self.hold:
                clock.now = max(clock.now, deadline)
            return None
        raw, delay = self.pending[0]
        if clock.now + delay > deadline:
            clock.now = max(clock.now, deadline)
            return None
        self.pending.pop(0)
        clock.now += delay
        self.world.frames.append(raw)
        return raw


class Pipe:
    def __init__(self) -> None:
        self.data = bytearray()
        self.shut = False
        self.position = 0


class Channel:
    def __init__(self, world: World, pipe: Pipe, stop: Callable[[], bool]) -> None:
        self.world, self.pipe, self.stop = world, pipe, stop

    def send_all(self, data: bytes) -> None:
        if self.stop() is not False:
            self.world.log.append(("send_stopped", len(data)))
            raise OSError("STOPPED")
        self.pipe.data.extend(data)


class Writer:
    def __init__(self, world: World, pipe: Pipe) -> None:
        self.world, self.pipe = world, pipe
        self.closed = False

    def channel(self, *, stop: Callable[[], bool]) -> Channel:
        self.world.log.append(("channel", stop() is False))
        return Channel(self.world, self.pipe, stop)

    def shutdown_write(self) -> None:
        self.world.log.append(("shutdown_write", len(self.pipe.data)))
        self.pipe.shut = True

    def close(self) -> None:
        self.world.log.append(("close_writer", self.closed))
        self.closed = True


class ReaderEnd:
    def __init__(self, world: World, pipe: Pipe) -> None:
        self.world, self.pipe = world, pipe
        self.closed = False

    def read(self, limit: int, *, deadline: float) -> bytes | None:
        self.world.log.append(("sink_read", limit, deadline))
        pipe = self.pipe
        if pipe.position < len(pipe.data):
            chunk = bytes(pipe.data[pipe.position : pipe.position + limit])
            pipe.position += len(chunk)
            return chunk
        if pipe.shut:
            return None
        raise TimeoutError

    def close(self) -> None:
        self.world.log.append(("close_reader", self.closed))
        self.closed = True


class Pair:
    def __init__(self, world: World) -> None:
        self.world = world
        self.count = 0

    def create(self) -> tuple[Writer, ReaderEnd]:
        self.count += 1
        self.world.log.append(("pair", self.count))
        if self.world.script.get("pair_fail") == self.count:
            raise OSError("PAIR_EFFECT_UNKNOWN")
        pipe = Pipe()
        writer, reader = Writer(self.world, pipe), ReaderEnd(self.world, pipe)
        self.world.endpoints += [writer, reader]
        return writer, reader


class Handle:
    def __init__(self, world: World, target: Callable[[], object], name: str) -> None:
        self.world, self.target, self.name = world, target, name
        self.started = False
        self.ran = False

    def start(self) -> bool:
        self.world.unlocked("start")
        self.world.log.append(("start", self.name))
        if self.world.script.get("start_fail") == self.name:
            return False
        self.started = True
        return True

    def run_now(self) -> None:
        if self.started and not self.ran:
            self.ran = True
            self.target()

    def join(self, *, deadline: float) -> bool:
        self.world.unlocked("join")
        self.world.log.append(("join", self.name, deadline))
        if self.world.script.get("join_fail") == self.name:
            return False
        self.run_now()
        return True


class Threads:
    def __init__(self, world: World) -> None:
        self.world = world

    def create(self, target: Callable[[], object], *, name: str) -> Handle:
        self.world.log.append(("create_thread", name))
        handle = Handle(self.world, target, name)
        self.world.handles.append(handle)
        return handle


class FakeSession:
    """Trusted stand-in of the public WorkerSession contract over the REAL ledger."""

    def __init__(self, world: World, index: int, token: Any, params: dict[str, Any]) -> None:
        self.world, self.index, self.token = world, index, token
        self.ledger, self.lock = params["ledger"], params["lock"]
        self.key = params["key"]
        self.state = SessionState.NEW
        self.jobs = 0
        self.snapshot = b""
        self.opened_at = 0.0

    def violation(self, text: str) -> None:
        self.world.violations.append(f"{self.index}:{text}")

    def open(self, snapshot: bytes) -> None:
        self.world.unlocked("open")
        self.world.log.append(("open", self.index))
        if self.state is not SessionState.NEW:
            self.violation("double-open")
        self.snapshot, self.opened_at = snapshot, self.world.clock.now
        mode = self.world.script.get("open_mode", {}).get(self.index)
        if mode == "new":
            raise WorkerSessionError(SessionFailure.OPEN)
        with self.lock:
            self.ledger.adopt_session(self.token)
        if mode == "held":
            self.cleanup(complete=False)
            raise WorkerSessionError(SessionFailure.OPEN)
        self.state = SessionState.READY

    def cleanup(self, *, complete: bool) -> ReleaseOutcome:
        with self.lock:
            self.ledger.begin_drain(self.token)
            outcome: ReleaseOutcome = self.ledger.release(
                self.token,
                CleanupConfirmation(
                    processes=True, threads=True, pipes=True, handles=True, output=complete
                ),
            )
        released = outcome is ReleaseOutcome.RELEASED
        self.state = SessionState.CLOSED if released else SessionState.HELD
        return outcome

    def query(self, sql: bytes) -> ErrorReply | None:
        self.world.unlocked("query")
        if self.state is not SessionState.READY or type(sql) is not bytes:
            self.violation("query-state")
        self.jobs += 1
        self.world.queries.append((self.index, sql, self.world.clock.now))
        if self.index == 0:
            return self.persistent(sql)
        if self.index == 1:
            self.cleanup(complete=True)
            return ErrorReply(self.key, 1, WireFailure.CONNECTION, False)
        if self.index == 4:
            self.heavy(sql)
        self.cleanup(complete=True)
        raise WorkerSessionError(SessionFailure.JOB)

    def persistent(self, sql: bytes) -> ErrorReply | None:
        job = self.jobs
        if sql != SQL[job].encode("ascii"):
            self.violation(f"sql-{job}")
        if job in ERRORS:
            return ErrorReply(self.key, job, WireFailure[ERRORS[job]], True)
        self.state = SessionState.RESULT
        return None

    def heavy(self, sql: bytes) -> None:
        if sql != HEAVY.encode("ascii"):
            self.violation("heavy-sql")
        if self.world.script.get("t4_timing", "during") == "during":
            for handle in self.world.handles:
                if handle.name == "watcher":
                    handle.run_now()

    def send_result_to(self, channel: Channel) -> ResultSummary:
        self.world.unlocked("send")
        job = self.jobs
        self.world.sends.append(job)
        if self.state is not SessionState.RESULT:
            self.violation("send-state")
        body = body_for(job)
        if self.world.script.get("corrupt_job") == job:
            body = body[:-1] + b"#"
        if self.world.script.get("late_send_job") == job:
            self.world.clock.now += 100.0
        failed = False
        try:
            for part in (body[: len(body) // 2], body[len(body) // 2 :]):
                channel.send_all(part)
        except OSError:
            failed = True
        if failed:
            self.cleanup(complete=True)
            raise WorkerSessionError(SessionFailure.OUTPUT)
        self.state = SessionState.READY
        return ResultSummary(1, rows_for(job), job == 9)

    def close(self) -> ReleaseOutcome:
        self.world.unlocked("close")
        self.world.log.append(("close", self.index))
        if self.state is SessionState.NEW:
            self.violation("close-new")
        if self.state is SessionState.CLOSED:
            return ReleaseOutcome.ALREADY_RELEASED
        return self.cleanup(complete=self.index not in self.world.script.get("close_held", ()))

    def request_stop(self) -> None:
        self.world.stops.append(self.index)


class Factory:
    def __init__(self, world: World) -> None:
        self.world = world

    def create(self, *args: Any, **kwargs: Any) -> FakeSession:
        world = self.world
        world.unlocked("create")
        index = len(world.tokens)
        world.log.append(("create", index))
        if len(args) != 2 or set(kwargs) != {"lock", "key", "limits", "profile", "guard"}:
            world.violations.append("create-signature")
        ledger, token = args
        world.ledger, world.lock = ledger, kwargs["lock"]
        world.tokens.append(token)
        world.keys.append(kwargs["key"])
        world.profiles.append(kwargs["profile"])
        world.limits.append(kwargs["limits"])
        if kwargs["guard"] is not GUARD_FACTS:
            world.violations.append("guard-identity")
        if index in world.script.get("create_fail", ()):
            raise WorkerSessionError(SessionFailure.STATE)
        session = FakeSession(world, index, token, {"ledger": ledger, **kwargs})
        world.sessions.append(session)
        return session


class World:
    def __init__(self, **script: Any) -> None:
        self.script = script
        self.clock = Clock()
        self.log: list[tuple[Any, ...]] = []
        self.violations: list[str] = []
        self.ledger: Any = None
        self.lock: Any = None
        self.armed_status: Any = None
        self.tokens: list[Any] = []
        self.keys: list[Any] = []
        self.profiles: list[Any] = []
        self.limits: list[Any] = []
        self.sessions: list[FakeSession] = []
        self.queries: list[tuple[int, bytes, float]] = []
        self.sends: list[int] = []
        self.stops: list[int] = []
        self.handles: list[Handle] = []
        self.endpoints: list[Any] = []
        self.frames: list[bytes] = []
        self.link = Link(self)

    def unlocked(self, where: str) -> None:
        if self.lock is not None and self.lock.locked():
            self.violations.append("ledger-lock-held:" + where)

    def run(self) -> Any:
        module = driver()
        ports = module["Ports"](
            reader=PureReader(self),
            guard=Guard(self),
            clock=self.clock,
            link=self.link,
            ids=Ids(self),
            factory=Factory(self),
            pair=Pair(self),
            threads=Threads(self),
        )
        return module["run"](ports)

    def events(self) -> list[dict[str, Any]]:
        return events_of(self.link.lines)

    def entries(self) -> list[Any]:
        return [row for row in self.ledger.status.entries if row is not None]

def result_event(case: int, seq: int, t4: int) -> dict[str, Any]:
    event: dict[str, Any] = {
        "bytes": None,
        "case": case,
        "e": "RESULT",
        "kind": "FATAL",
        "next_seq": seq,
        "reason": "JOB",
        "release": "ALREADY_RELEASED",
        "rows": None,
        "sink": None,
        "stop_applied": None,
        "truncated": None,
        "verdict": "PASS",
    }
    if case in ERRORS:
        event.update(kind="ERROR", reason=ERRORS[case], release=None)
    elif case <= 19:
        event.update(
            bytes=len(body_for(case)),
            kind="RESULT",
            reason=None,
            release=None,
            rows=rows_for(case),
            sink="OK",
            truncated=case == 9,
        )
    elif case == T1:
        event.update(reason="CONNECTION")
    elif case == T4:
        event.update(stop_applied=t4 == STOP, verdict="PASS" if t4 == STOP else "FAIL")
    elif case == T5:
        event.update(kind="PENDING", reason="T5_PENDING", release=None, verdict="FAIL")
    return event


def expected_walk(t4: int) -> list[dict[str, Any]]:
    events: list[dict[str, Any]] = [GUARD_EVENT, {"e": "INIT_OK"}]
    seq = 1
    for case in range(1, 25):
        events.append({"case": case, "e": "READY", "next_seq": seq})
        seq += 1
        if case == T4:
            events.append({"case": T4, "e": "ARMED", "live_window_ms": None, "next_seq": seq})
            seq += 1
        events.append(result_event(case, seq, t4))
        seq += 1
    failed = [T5] if t4 == STOP else [T4, T5]
    events.append({"e": "SUMMARY", "failed": failed, "global_pass": False, "prefix_cases": 24})
    events.append({"case": T6, "e": "READY", "next_seq": seq})
    events.append({"case": T6, "e": "ARMED", "live_window_ms": 28000, "next_seq": 51})
    events.append({"e": "TERMINAL", "outcome": "T6_UNKILLED"})
    return events

class Scripted:
    def __init__(self, clock: Clock, steps: list[object]) -> None:
        self.clock, self.steps = clock, steps
        self.limits: list[int] = []

    def read(self, limit: int, *, deadline: float) -> bytes | None:
        self.limits.append(limit)
        step = self.steps.pop(0) if self.steps else None
        if isinstance(step, BaseException):
            raise step
        if isinstance(step, float):
            self.clock.now = max(self.clock.now, deadline) + step
            return b"x"
        if step == "fill":
            return b"y" * limit
        if step is None or isinstance(step, bytes):
            return step
        return b"z" * (limit + 1)

        pytest.param([b"abcd", b"efX", None], b"abcdefg", "MISMATCH", 7, [4, 4, 2], id="hash"),
        pytest.param([b"abcd", None], b"abcdefg", "MISMATCH", 4, [4, 4], id="short"),
        pytest.param(["fill"] * 9, b"abcdefg", "OVERRUN", 9, [4, 4, 1], id="overrun"),

        pytest.param([OSError()], b"abcdefg", "IO", 0, [4], id="io"),
        pytest.param(["oversize"], b"abcdefg", "IO", 0, [4], id="oversize-chunk"),
        pytest.param([1.0, b"never"], b"abcdefg", "TIMEOUT", 1, [4], id="deadline"),
    ],
)
def test_d1_7_sink_drain_is_bounded_and_stops_at_cap(
