"""Main T5 probe binding (M5-1..7) over the closed fake System; RED-first, no effect.

Contract for the FUTURE amendment of ``deploy/oracle9_w1_main.py`` (T5-DESIGN v3 §3,
T5-DESIGN-REVIEW, T5-SOURCE-REVIEW). The current main (372f342f) still pins driver
39a68904 and has 20 System fields / 5 slots, so feature cases are expected to fail
first (old pin refused by ``load_driver``, missing ``ProbeChildren``/fields). That
API/pin RED proves no binding behavior. Main is loaded lazily (``m1.main()``) inside
feature cases only; ``test_sanity_*`` never loads it.

Harness: the reviewed ``FakeOS``/``FakeSock`` of ``tests/test_oracle9_w1_main.py`` are
extended by subclass (``ProbeOS``/``ProbeSock``); the old module and its globals are
not modified. ``ProbeOS.system`` builds the prospective 22-field System. One shared
``d1.Clock``. Nothing global (stdlib, environment, SDK algorithm) is patched.

Prospective stable contract fixed by this file:

- ``System`` has exactly the old 20 fields plus ``popen`` and ``pipe2``;
  ``real_system().popen is subprocess.Popen`` (never invoked) and ``pipe2`` is the
  module function ``_pipe2(flags)`` whose body only returns ``os.pipe2(flags)``; that
  is the only ``os.pipe2`` reference, so import/``real_system()`` never look it up.
- ``SLOTS`` = old five + ``probe_parent``, ``probe_child``, ``probe_stdin_r``,
  ``probe_stdin_w``, ``probe_process``; ``DRIVER_SHA``/``DRIVER_BYTES`` = b54a8a94/51913.
- ``build_ports`` passes ``children=ProbeChildren(root)``; ``SessionIds.nonce()``
  returns exactly 32 bytes from ``System.urandom(32)``, else ``MainRefusal("IDS")``.
- ``ProbeChildren(root).pair()``: both parent/child slots reserved UNKNOWN before
  ``System.socketpair(1, SOCK_STREAM)``, owned right after, then each socket
  ``set_inheritable(False)``. Parent facade: ``recv_exact(n, *, deadline)`` through the
  public ``RawChannel`` with ``ResultDeadline(expires_at=deadline, clock=System.monotonic)``
  and ``close() -> bool``; no send/write/shutdown/channel attribute. Child facade:
  ``fd`` and ``close() -> bool``. A socket close is positive only when ``fileno() == -1``.
- ``stdin()``: both stdin slots reserved UNKNOWN before ``System.pipe2(0o2000000)``
  (Linux O_CLOEXEC literal), owned right after, then only the writer
  ``set_blocking(w, False)``. Reader facade: ``fd``, ``close()``; writer facade:
  ``write_all(packet, *, deadline) -> bool`` (select/os_write, partial writes and EAGAIN
  retried, never a write after the absolute deadline) and ``close()``. A raw fd close is
  positive when ``System.close(fd)`` returns (os.close contract); OSError keeps the slot.
- ``spawn(child, reader, *, parent_pid, address_bytes, idle_ms)``: only the owned child
  and reader facades; process slot reserved UNKNOWN before ``System.popen`` with the
  installed worker argv, ``stdin=<reader fd>``, ``stdout/stderr=subprocess.DEVNULL``,
  ``env=child_environment()``, ``close_fds=True``, ``pass_fds=(<child fd>,)``,
  ``bufsize=0``, no shell; owned right after. Process facade: ``pid``,
  ``wait(*, deadline) -> int | None`` (finite raw ``wait(timeout)`` slices of at most
  ``SLICE_S``; only an exact int result clears the slot) and ``kill() -> bool``
  (never clears).

Unproven by these fakes (native Oracle proof): kernel EOF, dup2/fd inheritance and
close-on-exec, waitpid/returncode, the installed worker env/guard, a common monotonic
clock and PostgreSQL backend closure. The fake child honours its argv and the public
codecs only; it is not the real worker.
"""

# ---------------------------------------------------------------- fake OS extension


class ProbeSock(m1.FakeSock):
    """Fake socket end; parent reads never see EOF from closing its copy of the child end."""

    def __init__(self, fake: ProbeOS, fd: int) -> None:
        super().__init__(fake, fd)
        self.inheritable: bool | None = None

    def set_inheritable(self, flag: bool) -> None:
        self.fake.enter("set_inheritable", self.fd, self.fd, flag)
        self.inheritable = flag

    def recv(self, size: int) -> bytes:
        if self.closed:
            raise OSError(errno.EBADF, "EBADF")
        if self.buffer:
            data = bytes(self.buffer[:size])
            del self.buffer[:size]
            return data
        self.fake.clock.now += self.timeout if self.timeout is not None else 0.25
        raise TimeoutError


class FakeProc:
    """Raw fake process: natural exit 0 only after the parent socket end was closed."""

    def __init__(self, fake: ProbeOS, parent: ProbeSock, argv: list[str]) -> None:
        self.fake, self.parent, self.argv = fake, parent, argv
        self.pid = PROC_PID
        self.stdin = None  # Popen(stdin=<fd>) creates no pipe of its own
        self.returncode: int | None = None

    def wait(self, timeout: float) -> Any:
        self.fake.enter("proc_wait", None, timeout)
        if self.fake.wait_script:
            item = self.fake.wait_script.pop(0)
            if isinstance(item, BaseException):
                raise item
            return item
        if self.returncode is not None:
            return self.returncode
        if self.parent.closed:
            self.returncode = 0
            return 0
        self.fake.clock.now += timeout
        raise subprocess.TimeoutExpired("SYNTHETIC", timeout)

    def kill(self) -> None:
        self.fake.enter("proc_kill", None)
        self.returncode = -9


class ProbeOS(m1.FakeOS):
    """Reviewed FakeOS plus the two prospective calls and pipe/process fakes."""

    def __init__(self, clock: d1.Clock | None = None) -> None:
        super().__init__(clock)
        self.pipes: dict[int, str] = {}
        self.last_pipe: tuple[int, int] = (-1, -1)
        self.pipe_data = bytearray()
        self.pipe_limit = 1 << 20
        self.pipe_eagain = 0
        self.pipe_blocked = False
        self.procs: list[FakeProc] = []
        self.popen_calls: list[tuple[list[str], dict[str, Any]]] = []
        self.wait_script: list[object] = []
        self.random_override: object = None

    def system(self, ns: dict[str, Any]) -> Any:
        return ns["System"](**{name: getattr(self, name) for name in FIELDS22})

    def socketpair(self, family: object, kind: object) -> tuple[ProbeSock, ProbeSock]:
        self.enter("socketpair", None, family, kind)
        first, second = ProbeSock(self, self.next_fd), ProbeSock(self, self.next_fd + 1)
        self.next_fd += 2
        first.peer, second.peer = second, first
        self.socks += [first, second]
        return first, second

    def pipe2(self, flags: int) -> tuple[int, int]:
        self.enter("pipe2", None, flags)
        read_fd, write_fd = self.next_fd, self.next_fd + 1
        self.next_fd += 2
        self.pipes[read_fd], self.pipes[write_fd] = "r", "w"
        self.last_pipe = (read_fd, write_fd)
        return read_fd, write_fd

    def os_write(self, fd: int, data: bytes | memoryview) -> int:
        if fd not in self.pipes:
            return super().os_write(fd, data)
        self.enter("pipe_write", fd, fd, len(data))
        self.check(self.pipes[fd] == "w", "write-to-read-end")
        if self.pipe_eagain:
            self.pipe_eagain -= 1
            raise BlockingIOError(errno.EAGAIN, "EAGAIN")
        chunk = bytes(data)[: self.pipe_limit]
        self.pipe_data.extend(chunk)
        return len(chunk)

    def select(
        self, readers: list[int], writers: list[int], errors: list[int], timeout: float
    ) -> tuple[list[int], list[int], list[int]]:
        if not any(fd in self.pipes for fd in writers):
            return super().select(readers, writers, errors, timeout)
        self.enter("pipe_select", None, tuple(writers), timeout)
        self.check(not readers and not errors and timeout > 0, "pipe-select-args")
        if self.pipe_blocked:
            self.clock.now += timeout
            return [], [], []
        return [], list(writers), []

    def close(self, fd: int) -> None:
        if fd not in self.pipes:
            super().close(fd)
            return
        self.enter("pipe_close", fd, fd)
        kind = self.pipes.pop(fd)
        if kind == "w":
            self.child_reads_init()

    def child_reads_init(self) -> None:
        """Fake child: full INIT plus stdin EOF => OPENED with the public HMAC binding."""
        if not self.procs or len(self.pipe_data) < wire.HEADER_BYTES:
            return
        data = bytes(self.pipe_data)
        kind, size = wire.parse_header(data[: wire.HEADER_BYTES])
        if kind is not wire.Kind.INIT or size != len(data) - wire.HEADER_BYTES:
            return
        init = wire.decode_init(data[wire.HEADER_BYTES :])
        tag = hmac.digest(init.nonce, init.snapshot, "sha256")
        self.procs[-1].parent.buffer.extend(wire.encode_opened(tag))

    def popen(self, argv: list[str], **kwargs: Any) -> FakeProc:
        self.enter("popen", None, tuple(argv))
        self.popen_calls.append((list(argv), dict(kwargs)))
        child_fd = int(argv[5])
        child = next(sock for sock in self.socks if sock.fd == child_fd)
        parent = child.peer
        assert isinstance(parent, ProbeSock)
        proc = FakeProc(self, parent, list(argv))
        self.procs.append(proc)
        parent.buffer.extend(wire.encode_guard(PROC_PID, int(argv[6]), int(argv[7])))
        return proc

    def urandom(self, size: int) -> bytes:
        if self.random_override is None:
            return super().urandom(size)
        self.enter("urandom", None, size)
        value: Any = self.random_override
        return value  # deliberately foreign value for the refusal cases

def ns() -> dict[str, Any]:
    return m1.main()


def probe_states(root: Any) -> dict[str, str]:
    return {name: root.slots[name].state.name for name in PROBE_SLOTS}


def states_with(**changes: str) -> dict[str, str]:
    states = dict.fromkeys(PROBE_SLOTS, "EMPTY")
    states.update(changes)
    return states


def baseline(fake: ProbeOS) -> Any:
    """Clean decided root (completed ABORTED, link retired): safe_exit holds first."""
    root = m1.clean_root(fake)
    assert ns()["safe_exit"](root) is True
    fake.log.clear()
    return root


def acquire(fake: ProbeOS, root: Any) -> tuple[Any, Any, Any, Any, Any, Any]:
    children = ns()["ProbeChildren"](root)
    parent, child = children.pair()
    reader, writer = children.stdin()
    proc = children.spawn(
        child, reader, parent_pid=1, address_bytes=WORKER_AS, idle_ms=IDLE_MS
    )
    return children, parent, child, reader, writer, proc

def test_sanity_probe_os_speaks_public_codecs_and_keeps_child_copy_separate() -> None:
    fake = ProbeOS()
    parent, child = fake.socketpair(1, socket.SOCK_STREAM)
    read_fd, write_fd = fake.pipe2(O_CLOEXEC)
    proc = fake.popen(worker_argv(child.fd), stdin=read_fd, pass_fds=(child.fd,))
    kind, size = wire.parse_header(bytes(parent.buffer[: wire.HEADER_BYTES]))
    assert kind is wire.Kind.GUARD_READY
    body = bytes(parent.buffer[wire.HEADER_BYTES : wire.HEADER_BYTES + size])
    assert wire.decode_guard(body) == (PROC_PID, 1, WORKER_AS)
    parent.buffer.clear()
    child.close()  # the parent's local copy: never an EOF for the parent end
    parent.settimeout(0.25)
    with pytest.raises(TimeoutError):
        parent.recv(1)
    key = wire.SessionKey(bytes(16), d1.DATASOURCE_BYTES, 2, 1)
    limits = ResultLimits(max_rows=1000, max_frame_bytes=d1.FRAME_LIMIT)
    packet = wire.encode_init(wire.SecretInit(key, limits, 1012.0, bytes(32), b"SNAP"))
    assert fake.select([], [write_fd], [], 0.25) == ([], [write_fd], [])
    assert fake.os_write(write_fd, packet) == len(packet)
    fake.close(write_fd)
    tag = hmac.digest(bytes(32), b"SNAP", "sha256")
    assert bytes(parent.buffer) == wire.encode_opened(tag)
    with pytest.raises(subprocess.TimeoutExpired):
        proc.wait(0.25)
    parent.close()
    assert proc.wait(0.25) == 0
    assert proc.stdin is None
    assert fake.violations == []

def test_m5_2_pair_reserves_before_owns_after_then_noninheritable() -> None:
    fake = ProbeOS()
    root = baseline(fake)
    fake.probes["socketpair"] = lambda: probe_states(root)
    fake.probes["set_inheritable"] = lambda: probe_states(root)
    parent, child = ns()["ProbeChildren"](root).pair()
    unknown = states_with(probe_parent="UNKNOWN", probe_child="UNKNOWN")
    owned = states_with(probe_parent="OWNED", probe_child="OWNED")
    assert fake.observed["socketpair"] == [unknown]
    assert fake.observed["set_inheritable"] == [owned, owned]
    assert m1.ops(fake, "socketpair") == [("socketpair", 1, socket.SOCK_STREAM)]
    first, second = fake.socks
    assert (first.inheritable, second.inheritable) == (False, False)
    assert root.slots["probe_parent"].value is parent
    assert root.slots["probe_child"].value is child
    assert child.fd == second.fd
    for name in NO_SEND:
        assert not hasattr(parent, name)
    assert ns()["safe_exit"](root) is False  # owned ends, not a missing decision
    assert parent.close() is True and child.close() is True
    assert probe_states(root) == states_with()
    assert ns()["safe_exit"](root) is True


def test_m5_2_stdin_pipe2_cloexec_and_only_writer_nonblocking() -> None:
    fake = ProbeOS()

def test_m5_3_popen_after_four_owned_ends_with_exact_worker_arguments() -> None:
    fake = ProbeOS()
    root = baseline(fake)
    children = ns()["ProbeChildren"](root)
    parent, child = children.pair()
    reader, writer = children.stdin()

    assert str(write_fd) not in argv
    assert proc.pid == PROC_PID
    assert fake.procs[0].stdin is None

@pytest.mark.parametrize("which", ["child", "reader"])
def test_m5_3_spawn_refuses_foreign_ends_before_any_process(which: str) -> None:
    fake = ProbeOS()
    root = baseline(fake)
    children = ns()["ProbeChildren"](root)
    _, child = children.pair()
    reader, _ = children.stdin()

# ---------------------------------------------------------------- M5-4 failures and controls


def raise_control() -> None:
    raise KeyboardInterrupt


@pytest.mark.parametrize(
    ("stage", "mechanism", "error", "expected"),
    [
        pytest.param(
            "pair",
            "socketpair",
            m1.FakeOSError(),
            {"probe_parent": "UNKNOWN", "probe_child": "UNKNOWN"},
            id="socketpair-error",
        ),
        pytest.param(
            "pair",
            "socketpair",
            KeyboardInterrupt(),
            {"probe_parent": "UNKNOWN", "probe_child": "UNKNOWN"},
            id="socketpair-control",
        ),
        pytest.param(
            "pair",
            "set_inheritable",
            None,
            {"probe_parent": "OWNED", "probe_child": "OWNED"},
            id="control-after-socketpair-return",
        ),
        pytest.param(
            "stdin",
            "pipe2",
            m1.FakeOSError(),
            {"probe_stdin_r": "UNKNOWN", "probe_stdin_w": "UNKNOWN"},
            id="pipe2-error",
        ),
        pytest.param(
            "stdin",
            "set_blocking",
            None,
            {"probe_stdin_r": "OWNED", "probe_stdin_w": "OWNED"},
            id="control-after-pipe2-return",
        ),
        pytest.param(
            "spawn", "popen", m1.FakeOSError(), {"probe_process": "UNKNOWN"}, id="popen-error"
        ),
        pytest.param(
            "spawn", "popen", KeyboardInterrupt(), {"probe_process": "UNKNOWN"}, id="popen-control"
        ),
    ],
)
def test_m5_4_creation_failure_or_control_retains_and_blocks_safe_exit(
    stage: str, mechanism: str, error: BaseException | None, expected: dict[str, str]
) -> None:
    fake = ProbeOS()
    root = baseline(fake)
    children = ns()["ProbeChildren"](root)
    prior: dict[str, str] = {}
    child: Any = None
    reader: Any = None
    if stage in ("stdin", "spawn"):
        _, child = children.pair()
        prior |= {"probe_parent": "OWNED", "probe_child": "OWNED"}
    if stage == "spawn":
        reader, _ = children.stdin()
        prior |= {"probe_stdin_r": "OWNED", "probe_stdin_w": "OWNED"}
    if error is None:
        fake.hooks[mechanism] = raise_control  # runs after the creating call returned
    else:
        fake.fail[mechanism] = {None: error}
    with pytest.raises((m1.FakeOSError, KeyboardInterrupt)):
        if stage == "pair":
            children.pair()
        elif stage == "stdin":
            children.stdin()
        else:
            children.spawn(child, reader, parent_pid=1, address_bytes=WORKER_AS, idle_ms=IDLE_MS)
    assert probe_states(root) == states_with(**prior, **expected)
    assert ns()["safe_exit"](root) is False
    for name, state in expected.items():
        if state == "OWNED":
            assert root.slots[name].value is not None  # the original stays reachable

@pytest.mark.parametrize("end", ["reader", "writer"])

    fake = ProbeOS()

@pytest.mark.parametrize(
    ("script", "reaped"),
    [
        pytest.param([0], 0, id="exact-zero"),
        pytest.param([1], 1, id="exact-one"),
        pytest.param([True], None, id="bool"),
        pytest.param([IntZero(0)], None, id="int-subclass"),
        pytest.param([0.0], None, id="float"),
        pytest.param([m1.FakeOSError()], None, id="oserror"),
    ],
)
def test_m5_6_only_exact_int_wait_result_clears_process(
    script: list[object], reaped: int | None
) -> None:
    fake = ProbeOS()
    root = baseline(fake)
    _, _, _, _, _, proc = acquire(fake, root)
    fake.wait_script = list(script)
    result = proc.wait(deadline=fake.clock.now + 1.0)
    if reaped is None:
        assert result is None
        assert root.slots["probe_process"].state.name == "OWNED"
        assert root.slots["probe_process"].value is proc
    else:
        assert type(result) is int and result == reaped
        assert root.slots["probe_process"].state.name == "EMPTY"


def test_m5_6_wait_uses_finite_slices_and_never_waits_after_deadline() -> None:
    fake = ProbeOS()
    root = baseline(fake)
    _, parent, _, _, _, proc = acquire(fake, root)
    deadline = fake.clock.now + 0.6
    assert proc.wait(deadline=deadline) is None  # no EOF yet: natural timeout
    timeouts = [entry[1] for entry in m1.ops(fake, "proc_wait")]
    assert timeouts and all(0 < value <= 0.25 for value in timeouts)
    assert fake.clock.now >= deadline
    fake.log.clear()
    assert proc.wait(deadline=fake.clock.now) is None  # expired before any raw wait
    assert m1.ops(fake, "proc_wait") == []
    assert root.slots["probe_process"].state.name == "OWNED"
    assert parent.close() is True
    assert proc.wait(deadline=fake.clock.now + 1.0) == 0
    assert root.slots["probe_process"].state.name == "EMPTY"


def test_m5_6_kill_alone_never_clears_the_process() -> None:

def test_m5_7_parent_receives_through_rawchannel_with_finite_deadline() -> None:
    fake = ProbeOS()
    root = baseline(fake)
    children = ns()["ProbeChildren"](root)
    parent, child = children.pair()
    first, _ = fake.socks
    first.buffer.extend(b"abcdef")
    assert parent.recv_exact(4, deadline=fake.clock.now + 1.0) == b"abcd"
    assert child.close() is True  # the parent's copy of the child end
    deadline = fake.clock.now + 0.6
    with pytest.raises(TransportError) as caught:
        parent.recv_exact(4, deadline=deadline)  # 2 bytes buffered, no EOF from that close

    fake.sock_lies.add(first.fd)
    assert parent.close() is False
    assert root.slots["probe_parent"].state.name == "OWNED"
    fake.sock_lies.clear()
    assert parent.close() is True
    assert probe_states(root) == states_with()

# ---------------------------------------------------------------- composition


def test_composition_public_driver_run_with_main_probe_binding() -> None:
    """Public driver.run + main ProbeChildren over fake OS; other cases use D1 fakes."""
    world = d1.World()
    fake = ProbeOS(clock=world.clock)
    root = m1.bound_root(fake)  # verified load_driver: same bytes, pinned SHA
    module = root.namespace
    owner = root.owner
    ports = module.Ports(
        reader=d1.PureReader(world),
        guard=d1.Guard(world),
        clock=world.clock,
        link=world.link,
        ids=ns()["SessionIds"](root),
        factory=RootFactory(world, root),
        pair=d1.Pair(world),
        threads=d1.Threads(world),
        children=ns()["ProbeChildren"](root),
    )
    result = module.run(ports, owner=owner)
    assert result.outcome == "T6_UNKILLED"
    assert result.verdicts[T5 - 1] == "PASS"
    events = d1.events_of(world.link.lines)
    eof = [event for event in events if event["e"] == "RESULT" and event["case"] == T5]
    assert eof == [d1.result_event(T5, 49, d1.STOP)]
    assert probe_states(root) == states_with()
    assert (owner.session, owner.session_token) == (None, None)
    assert (owner.endpoints, owner.handles, owner.unknown) == ((), (), ())
    assert len(fake.procs) == 1 and fake.procs[0].returncode == 0
    assert m1.ops(fake, "proc_kill") == []
    assert all(sock.closed for sock in fake.socks) and fake.pipes == {}
    argv, kwargs = fake.popen_calls[0]
    assert kwargs["pass_fds"] == (int(argv[5]),) and kwargs["stdin"] == fake.last_pipe[0]
    data = bytes(fake.pipe_data)
    assert wire.parse_header(data[: wire.HEADER_BYTES]) == (
        wire.Kind.INIT,
        len(data) - wire.HEADER_BYTES,
    )
    init = wire.decode_init(data[wire.HEADER_BYTES :])
    assert (init.key.generation, len(init.nonce)) == (2, 32)
    assert init.snapshot == world.sessions[4].snapshot  # same G2 snapshot as T4/T6
    assert m1.ops(fake, "urandom").count(("urandom", 32)) == 1
    assert root.ledger is world.ledger
    rows = [row for row in world.ledger.status.entries if row is not None]
    assert not any(row.kind is ReservationKind.SESSION for row in rows)
    assert world.violations == [] and fake.violations == []
    root.result = result
    assert ns()["safe_exit"](root) is True  # every probe resource settled by real facts
