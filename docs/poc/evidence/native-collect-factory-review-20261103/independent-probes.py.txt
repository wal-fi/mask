import ctypes,json,pytest,runpy
h=runpy.run_path('tests/test_pg16_shell_native_control_collect_factory.py')
with pytest.MonkeyPatch.context() as mp:
    pin=h['pin'].__wrapped__(mp)
    world=h['world'](pin)
    calls=[]
    def factory(ns):
        def refuse_create(*args):
            assert len(args)==11
            ns['PROTOTYPES']['NtCreateFile'][1][2].from_param(args[2])
            ns['PROTOTYPES']['NtCreateFile'][1][3].from_param(args[3])
            calls.append(True)
            return -1073741790
        def forbidden(*args):
            raise AssertionError('unexpected simulated delegate')
        bound={name:forbidden for name in ns['PROTOTYPES']}
        bound['NtCreateFile']=refuse_create
        return ns['_NativeApi'](bound)
    with h['sealed'](pin):
        result=h['run'](pin,factory)
    assert result==('INVALID','COLLECT_OPEN',None,None),result
    record=pin.glob['_ATTEMPTS'][0]
    assert record.factory is factory and record.factory_started is True
    assert len(calls)==1
    assert type(record.api) is record.native['_NativeApi']
    assert record.backend._open=={} and record.backend._retained==[]
    assert not record.collector['_ACTIVE'] and not record.collector['_POISON']
    bridge={'selected_namespace_types_accepted':True,'python_delegate_count':len(calls),'closed_refusal':result[:2]}
with pytest.MonkeyPatch.context() as mp:
    pin=h['pin'].__wrapped__(mp)
    world=h['world'](pin)
    factory=h['Factory'](pin,result=world.api)
    refused=False
    with h['sealed'](pin) as patch:
        h['no_exec'](patch)
        try:
            pin.run(pin.collector_src,pin.backend_src,None,h['NONCE'],h['ARGV0'],factory)
        except TypeError:
            refused=True
    assert refused and not factory.calls and pin.glob['_ATTEMPTS']==[] and world.nt.calls==[]
print(json.dumps({'fake_native_api_bridge':bridge,'keyword_only_refused_before_effect':refused,'dll_loader_or_native_API_called':False},sort_keys=True))
