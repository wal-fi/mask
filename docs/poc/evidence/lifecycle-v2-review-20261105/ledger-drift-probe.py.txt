with pytest.MonkeyPatch.context() as patch:
    lab=suite['lab'].__wrapped__(patch)
    owner,kernel=suite['owner_for'](lab,'NORMAL');suite['tree'](kernel)
    case=suite['case_for'](lab,'NORMAL',1,{})
    seen={}
    def changed(target,facts):
        seen['original_ledger_len']=len(target.ledger)
        target.ledger={}
        seen['after_calls']=len(kernel.calls)

    ctx=v2['context'](owner,case,lab.runner,changed)

    print(json.dumps({'ledger_drift_probe':seen}))
