class FakeFailure(OSError):
    """Synthetic ordinary failure of a fake OS call; carries no data."""

    def __init__(self) -> None:
        super().__init__(errno.EIO, "SYNTHETIC")


class Reactor:
    """D1-style host: GO after READY, ACK after RESULT; configured EOF instead of GO."""

    def __init__(self, fake: FakeOS, eof: set[tuple[int, int]]) -> None:
        self.fake, self.eof = fake, eof

    def react(self, line: bytes) -> None:
        event = json.loads(line)
        kind, case, seq = event["e"], event.get("case", 0), event.get("next_seq", 0)
        if kind == "READY":
            self.queue(d1.GO, case, seq)
        elif kind == "RESULT":
            self.queue(d1.ACK, case, seq)

    def queue(self, kind: int, case: int, seq: int) -> None:
        if (kind, case) in self.eof:
            self.fake.eof = True
            return
        self.fake.inbound.extend(d1.frame(kind, case, seq))


class FakeThread:
    def __init__(
        self, fake: FakeOS, target: Callable[[], object], name: str, daemon: bool
    ) -> None:
        self.fake, self.target, self.name, self.daemon = fake, target, name, daemon
        self.started = False
        self.alive = False
        self.runs = 0

    def start(self) -> None:
        self.fake.enter("thread_start", self.name, self.name)
        self.started = True
        self.alive = True
        if self.name in self.fake.finish_on_start:
            self.run_target()

    def run_target(self) -> None:
        previous = self.fake.main
        self.fake.main = False  # the target runs in a non-main role
        self.runs += 1
        try:
            self.target()
        finally:
            self.fake.main = previous
            self.alive = False

    def join(self, timeout: float | None = None) -> None:
        self.fake.enter("thread_join", None, self.name, timeout)
        failure = self.fake.join_raise.get(self.name)
        if failure is not None:
            self.alive = False  # would look closed afterwards; the join itself failed
            raise failure
        if self.name in self.fake.stuck or (not self.alive and self.runs == 0):
            # Live-but-stuck or never-entered thread: the join slice elapses, nothing runs.
            self.fake.clock.now += timeout if timeout is not None else 0.25
            return
        if self.alive and self.runs == 0:
            self.run_target()

    def is_alive(self) -> bool:
        return self.alive


class FakeSock:
    def __init__(self, fake: FakeOS, fd: int) -> None:
        self.fake, self.fd = fake, fd
        self.peer: FakeSock | None = None
        self.buffer = bytearray()
        self.write_shut = False
        self.closed = False
        self.timeout: float | None = None

    def settimeout(self, value: float | None) -> None:
        self.fake.check(value is None or value > 0, "settimeout")
        self.timeout = value

    def send(self, view: memoryview) -> int:
        peer = self.peer
        if self.closed or self.write_shut or peer is None:
            raise OSError(errno.EPIPE, "EPIPE")
        peer.buffer.extend(view)
        return len(view)

    def recv(self, size: int) -> bytes:
        if self.closed:
            raise OSError(errno.EBADF, "EBADF")
        if self.buffer:
            data = bytes(self.buffer[:size])
            del self.buffer[:size]
            return data
        peer = self.peer
        if peer is not None and (peer.write_shut or peer.closed):
            return b""
        self.fake.clock.now += self.timeout if self.timeout is not None else 0.25
        raise TimeoutError

    def shutdown(self, how: int) -> None:
        self.fake.enter("sock_shutdown", self.fd, self.fd, how)
        self.write_shut = True

    def close(self) -> None:
        self.fake.enter("sock_close", self.fd, self.fd)
        if self.fd not in self.fake.sock_lies:
            self.closed = True

    def fileno(self) -> int:
        return -1 if self.closed else self.fd


class FakeOS:
    """Records every call; failures, probes and one-shot hooks are per operation."""

    def __init__(self, clock: d1.Clock | None = None) -> None:
        self.clock = clock if clock is not None else d1.Clock()
        self.log: list[tuple[Any, ...]] = []
        self.violations: list[str] = []
        self.foreign: list[tuple[str, int]] = []
        self.fail: dict[str, dict[object, BaseException]] = {}
        self.probes: dict[str, Callable[[], object]] = {}
        self.observed: dict[str, list[object]] = {}
        self.hooks: dict[str, Callable[[], None]] = {}
        self.files: dict[str, bytes] = {DRIVER_PATH: DRIVER_SOURCE, PURE_PATH: PURE_SOURCE}
        self.modes: dict[str, int] = {}
        self.lies: set[str] = set()
        self.open_fds: dict[int, str] = {}
        self.positions: dict[int, int] = {}
        self.opened: list[tuple[str, int]] = []
        self.next_fd = 20
        self.file_chunk = 1 << 20
        self.inbound = bytearray()
        self.chunk = 1 << 20
        self.eof = False
        self.read_late = 0.0
        self.select_late = 0.0
        self.fd_modes = {0: stat.S_IFIFO, 1: stat.S_IFIFO}
        self.ttys: set[int] = set()
        self.write_limit = 1 << 20
        self.fd1_blocked = False
        self.outbound = bytearray()
        self.lines: list[bytes] = []
        self.reactor: Reactor | None = None
        self.handlers: dict[int, Any] = {}
        self.main = True
        self.threads: list[FakeThread] = []
        self.finish_on_start: set[str] = set()
        self.stuck: set[str] = set()
        self.join_raise: dict[str, BaseException] = {}
        self.socks: list[FakeSock] = []
        self.sock_lies: set[int] = set()
        self.extra_threads = 0
        self.randoms = 0
        self.facts: GuardFacts = d1.GUARD_FACTS

    def check(self, condition: bool, text: str) -> None:
        if not condition:
            self.violations.append(text)

    def enter(self, name: str, key: object, *logged: object) -> None:
        self.log.append((name, *logged))
        probe = self.probes.get(name)
        if probe is not None:
            self.observed.setdefault(name, []).append(probe())
        hook = self.hooks.pop(name, None)
        if hook is not None:
            hook()
        failure = self.fail.get(name, {}).get(key)
        if failure is not None:
            raise failure

    def system(self, ns: dict[str, Any]) -> Any:
        return ns["System"](**{name: getattr(self, name) for name in SYSTEM_FIELDS})

    def open(self, path: str, flags: int) -> int:
        self.check(type(path) is str and type(flags) is int, "open-args")
        self.enter("open", path, path)
        fd = self.next_fd
        self.next_fd += 1
        self.open_fds[fd] = path
        self.positions[fd] = 0
        self.opened.append((path, fd))
        return fd

    def read(self, fd: int, size: int) -> bytes:
        path = self.open_fds.get(fd)
        self.enter("read", path, fd, size)
        if path is None:
            raise OSError(errno.EBADF, "EBADF")
        start = self.positions[fd]
        piece = self.files[path][start : start + min(size, self.file_chunk)]
        self.positions[fd] = start + len(piece)
        return piece

    def close(self, fd: int) -> None:
        path = self.open_fds.get(fd)
        self.check(fd not in (0, 1, 2), "close-borrowed")
        self.enter("close", path, fd)
        if path is not None and path not in self.lies:
            del self.open_fds[fd]
            del self.positions[fd]

    def fstat(self, fd: int) -> SimpleNamespace:
        path = self.open_fds.get(fd)
        self.enter("fstat", fd, fd)
        if path is not None:
            mode = self.modes.get(path, stat.S_IFREG)
            return SimpleNamespace(st_mode=mode | 0o444, st_size=len(self.files[path]))
        if fd in self.fd_modes:
            return SimpleNamespace(st_mode=self.fd_modes[fd] | 0o600, st_size=0)
        raise OSError(errno.EBADF, "EBADF")

    def isatty(self, fd: int) -> bool:
        self.enter("isatty", fd, fd)
        return fd in self.ttys

    def set_blocking(self, fd: int, flag: bool) -> None:
        self.enter("set_blocking", fd, fd, flag)

    def os_read(self, fd: int, size: int) -> bytes:
        self.enter("os_read", fd, fd, size)
        if fd != 0:
            self.foreign.append(("os_read", fd))
            raise OSError(errno.EBADF, "EBADF")
        if self.inbound:
            count = min(size, self.chunk, len(self.inbound))
            data = bytes(self.inbound[:count])
            del self.inbound[:count]
            self.clock.now += self.read_late
            return data
        if self.eof:
            return b""
        raise BlockingIOError(errno.EAGAIN, "EAGAIN")

    def os_write(self, fd: int, data: bytes | memoryview) -> int:
        self.enter("os_write", fd, fd, len(data))
        if fd != 1:
            self.foreign.append(("os_write", fd))
            return len(data)
        chunk = bytes(data)[: self.write_limit]
        self.outbound.extend(chunk)
        while b"\n" in self.outbound:
            line, _, rest = bytes(self.outbound).partition(b"\n")
            self.outbound = bytearray(rest)
            self.lines.append(line)
            if self.reactor is not None:
                self.reactor.react(line)
        return len(chunk)

    def select(
        self, readers: list[int], writers: list[int], errors: list[int], timeout: float
    ) -> tuple[list[int], list[int], list[int]]:
        self.enter("select", None, tuple(readers), tuple(writers), timeout)
        self.check(not errors and timeout > 0, "select-args")
        ready_r = [0] if 0 in readers and (self.inbound or self.eof) else []
        ready_w = [1] if 1 in writers and not self.fd1_blocked else []
        if ready_r or ready_w:
            self.clock.now += self.select_late
        else:
            self.clock.now += timeout
        return ready_r, ready_w, []

    def socketpair(self, family: object, kind: object) -> tuple[FakeSock, FakeSock]:
        self.enter("socketpair", None, family, kind)
        first, second = FakeSock(self, self.next_fd), FakeSock(self, self.next_fd + 1)
        self.next_fd += 2
        first.peer, second.peer = second, first
        self.socks += [first, second]
        return first, second

    def thread_class(
        self, *, target: Callable[[], object], name: str, daemon: bool
    ) -> FakeThread:
        self.enter("thread_class", name, name, daemon)
        thread = FakeThread(self, target, name, daemon)
        self.threads.append(thread)
        return thread

    def active_count(self) -> int:
        return 1 + sum(thread.alive for thread in self.threads) + self.extra_threads

    def signal(self, signum: int, handler: object) -> None:
        self.enter("signal", signum, signum)
        self.handlers[signum] = handler

    def monotonic(self) -> float:
        return self.clock.now

    def sleep(self, seconds: float) -> None:
        self.enter("sleep", None, seconds)
        self.clock.now += seconds

    def pause(self) -> None:
        self.enter("pause", None)

    def urandom(self, size: int) -> bytes:
        self.enter("urandom", None, size)
        self.randoms += 1
        return bytes([0xA0 + self.randoms]) * size

    def protect(self, *, role: object, address_limit_bytes: int) -> GuardFacts:
        self.enter("protect", None, role, address_limit_bytes)
        return self.facts

    def is_main_thread(self) -> bool:
        return self.main

    def silence(self) -> None:
        self.enter("silence", None)

def assert_closed(error: BaseException, cls: type[BaseException]) -> None:
    assert type(error) is cls
    assert error.__cause__ is None
    assert error.__context__ is None
    for leak in LEAKS:
        assert leak not in str(error) + repr(error)


def assert_within_frame(requests: list[int], parts: list[int], chunk: int) -> None:
    """Every os_read asks for at most the bytes still missing in the current part."""
    remaining = list(parts)
    index = 0
    for size in requests:
        while index < len(remaining) and remaining[index] == 0:
            index += 1
        assert index < len(remaining)
        assert 0 < size <= remaining[index]
        remaining[index] -= min(size, chunk)
    assert remaining == [0] * len(parts)


def lab_ledger(*, session: bool) -> BudgetLedger:
    ledger = BudgetLedger(s2.LAB_BUDGET, (DatasourceCapacity(1, 1),))
    generation = ledger.reserve_candidate(0, shared_bytes=MIB)
    ledger.publish(generation)
    if session:
        ledger.reserve_session(
            generation, worker_bytes=256 * MIB, parent_bytes=s2.PARENT_BYTES
        )
    return ledger


def lab_token() -> tuple[BudgetLedger, Any]:
    ledger = BudgetLedger(s2.LAB_BUDGET, (DatasourceCapacity(1, 1),))

def clean_root(fake: FakeOS) -> Any:
    root, link = linked(fake)
    assert link.retire() is True
    root.result = root.namespace.DriverResult("ABORTED", ("NOT_RUN",) * 25, None)
    return root


def forge(**fields: object) -> Any:
    decision = object.__new__(main()["MainDecision"])
    for name, value in fields.items():
        object.__setattr__(decision, name, value)
    return decision


def main_tree() -> ast.Module:

def function(tree: ast.Module, name: str) -> ast.FunctionDef:
    found = [node for node in tree.body if isinstance(node, ast.FunctionDef) and node.name == name]
    assert len(found) == 1
    return found[0]


def argument_names(node: ast.FunctionDef) -> list[str]:
    args = node.args
    assert args.vararg is None
    assert args.kwarg is None
    assert args.defaults == []
    return [arg.arg for arg in [*args.posonlyargs, *args.args, *args.kwonlyargs]]


def calls_in(node: ast.AST) -> list[str]:
    return [ast.unparse(item) for item in ast.walk(node) if isinstance(item, ast.Call)]

    )

def test_m01_ast_top_level_has_only_definitions_and_one_main_guard() -> None:
    tree = main_tree()
    guards: list[ast.If] = []
    modules: set[str] = set()
    definitions = (ast.Assign, ast.AnnAssign, ast.FunctionDef, ast.ClassDef)
    for node in tree.body:
        if isinstance(node, ast.Expr) and isinstance(node.value, ast.Constant):
            continue
        if isinstance(node, ast.If):
            guards.append(node)
        elif isinstance(node, ast.Import):
            modules.update(alias.name for alias in node.names)
        elif isinstance(node, ast.ImportFrom):
            assert node.level == 0
            modules.add(str(node.module))
        else:
            assert isinstance(node, definitions)
            if isinstance(node, (ast.Assign, ast.AnnAssign)):
                called = {call.split("(")[0] for call in calls_in(node)}
                assert not called & {"entry", "_run_once", "real_system", "hold", "hold_step"}
    assert modules <= ALLOWED_IMPORTS
    assert len(guards) == 1
    guard = guards[0]
    assert ast.unparse(guard.test) == "__name__ == '__main__'"
    assert [ast.unparse(item) for item in guard.body] == ["raise SystemExit(entry(real_system()))"]
    assert guard.orelse == []
    texts = {
        node.value
        for node in ast.walk(tree)
        if isinstance(node, ast.Constant) and isinstance(node.value, str)
    }
    assert "MAIN" not in texts  # no main-owned protocol event
    source = MAIN_FILE.read_text(encoding="utf-8")
    for absent in ("rounds", "real_primitives", "print("):
        assert absent not in source

def test_m20_ast_hold_never_returns_and_entry_retains_its_root() -> None:
    tree = main_tree()

    assert argument_names(hold) == ["root"]
    body = [
        node
        for node in hold.body
        if not (isinstance(node, ast.Expr) and isinstance(node.value, ast.Constant))
    ]
    assert len(body) == 1
    loop = body[0]
    assert isinstance(loop, ast.While)
    assert isinstance(loop.test, ast.Constant)

    assert loop.orelse == []
