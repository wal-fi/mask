def reactor_variant(**changes: object) -> types.ModuleType:
    """The reviewed reactor namespace bound into a plain ModuleType (caller-trusted)."""
    module = types.ModuleType("_duplex_host_reactor")
    module.__dict__.update(ht._reactor_namespace())
    for name, value in changes.items():
        if value is ht.DROP:
            del module.__dict__[name]
        else:
            module.__dict__[name] = value
    return module

def respond(payload: bytes, live: int = ht.LIVE_MS) -> list[dict[str, object]]:
    """Controller lines answering one accepted host write (b54 happy handshake)."""
    if payload == PACKET:
        return [ht.INIT_OK, ht.ready(1, 1)]
    _, _, kind, case, _, _ = d1.CONTROL.unpack(payload)
    if kind == GO:
        armed = {23: ht.armed(23, 46, None), 25: ht.armed(25, 51, live)}
        return [armed[case] if case in armed else ht.result(case)]
    if kind in (STOP, NOSTOP):
        return [ht.result(23)]
    if case == 24:
        return [ht.summary([]), ht.ready(25, 50)]
    return [ht.ready(case + 1, ht.READY_SEQ[case + 1])]

@dataclasses.dataclass(frozen=True)
class Replace:
    """Fault plan: the op completes normally, but outcome() returns exactly this value."""

    value: object


class Op:
    """Inert fake operation: completes only when the logical clock reaches its event."""

    def __init__(self, world: World, role: str) -> None:
        self.world = world
        self.role = role
        self.ready_at: float | None = None
        self.result: object = None
        self.fault: object = None
        self.harvested = False
        self.done_times: list[float] = []
        self.outcome_calls = 0
        world.ops.append(self)

    def next_time(self) -> float | None:
        return self.ready_at

    def make(self) -> object:
        return self.result

    def done(self) -> object:
        w = self.world
        w.trace.append(("done", self.role, w.t))
        if self.harvested:
            w.violations.append("done-after-harvest:" + self.role)
        self.done_times.append(w.t)
        if self.fault == "done_raise":
            w.mark()
            raise RuntimeError("synthetic done fault")
        if self.fault == "done_int":
            w.mark()
            return 1
        when = self.next_time()
        return when is not None and when <= w.t

    def outcome(self) -> object:
        w = self.world
        w.trace.append(("outcome", self.role, w.t))
        when = self.next_time()
        if self.harvested or when is None or when > w.t:
            w.violations.append("outcome-twice-or-before-done:" + self.role)
        self.harvested = True
        self.outcome_calls += 1
        if self.fault == "outcome_raise":
            w.mark()
            raise RuntimeError("synthetic outcome fault")
        if isinstance(self.fault, BaseException):
            w.mark()
            raise self.fault
        if isinstance(self.fault, Replace):
            return self.fault.value
        return self.make()


class ReadOp(Op):
    def __init__(self, world: World, limit: object) -> None:
        super().__init__(world, "read")
        self.limit = limit

    def next_time(self) -> float | None:
        return self.world.readable_at()

    def make(self) -> Any:
        return self.world.b.ReadOutcome(ok=True, data=self.world.take(self.limit))


class WaitOp(Op):
    def __init__(self, world: World) -> None:
        super().__init__(world, "wait")

    def next_time(self) -> float | None:
        return self.world.exit_at

    def make(self) -> object:
        return self.world.exit_outcome


class Clock:
    """Logical clock: poll returns at the first new completion or at the deadline."""

    def __init__(self, world: World) -> None:
        self.world = world
        self.overshoot = False
        self.on_now: Any = None
        self.on_poll: Any = None
        self.bridge: Any = None

    def now(self) -> Any:
        w = self.world
        w.trace.append(("now", w.t))
        if self.on_now is not None:
            return self.on_now(self)
        return w.t

    def poll(self, watched: Any, deadline: Any) -> Any:
        w = self.world
        names = tuple(getattr(slot, "name", None) for slot in watched)
        w.trace.append(("poll", names, deadline, w.t))
        ops = self.check(watched)
        if self.on_poll is not None:
            forced = self.on_poll(self, watched, deadline)
            if forced is not None:
                return forced
        times = [when for op in ops if (when := op.next_time()) is not None]
        upcoming = min(times) if times else None
        reached = upcoming is not None and (upcoming <= max(w.t, deadline) or self.overshoot)
        target = upcoming if reached and upcoming is not None else deadline
        w.t = max(w.t, target)
        return w.t

    def check(self, watched: Any) -> list[Op]:
        """R7.2/R8.3: only occupied, never-harvested slots; never an empty tuple."""
        w = self.world
        if type(watched) is not tuple or not watched:
            w.violations.append("poll-without-eligible-slot")
            return []
        ops: list[Op] = []
        for slot in watched:
            op = slot.op if type(slot) is w.b.OperationSlot else None
            if not isinstance(op, Op) or slot.occupied is not True or op.harvested:
                w.violations.append("poll-watches-ineligible-slot")
            else:
                ops.append(op)
        return ops


class Gateway:
    def __init__(self, world: World) -> None:
        self.world = world

    def begin_spawn_into(self, slot: Any, deadline: Any) -> None:
        w = self.world
        w.trace.append(("spawn", deadline, w.t))
        op = Op(w, "spawn")
        w.attach("spawn", slot, op)
        op.ready_at, op.result = w.on_spawn(w.t)


class Process:
    def __init__(self, world: World) -> None:
        self.world = world

    def begin_read_into(self, slot: Any, max_bytes: Any, deadline: Any) -> None:
        w = self.world
        w.trace.append(("read", max_bytes, deadline, w.t, w.consumed))
        w.attach("read", slot, ReadOp(w, max_bytes))

    def begin_write_into(self, slot: Any, payload: Any, deadline: Any) -> None:
        w = self.world
        w.trace.append(("write", payload, deadline, w.t, w.reactor.status().pending))
        op = Op(w, "write")
        w.attach("write", slot, op)
        op.ready_at, op.result = w.on_write(payload, w.t, deadline)

    def begin_wait_into(self, slot: Any, deadline: Any) -> None:
        w = self.world
        w.trace.append(("wait", deadline, w.t))
        w.attach("wait", slot, WaitOp(w))


class Acts:
    def __init__(self, world: World) -> None:
        self.world = world

    def begin_act_into(self, slot: Any, act: Any, deadline: Any) -> None:
        w = self.world
        pending = w.reactor.status().pending
        w.trace.append(("act", act.kind.name, act.case, deadline, w.t, act, pending))
        op = Op(w, "act")
        w.attach("act", slot, op)
        op.ready_at, op.result = w.on_act(act, w.t, deadline)


class World:
    """One logical dyadic clock, the fake gateway stdout/exit and every fake operation."""

    def __init__(self, b: Any, r: Any, script: Script) -> None:
        self.b = b
        self.r = r
        self.script = script
        self.t = T0
        self.overall = T0 + OVERALL
        self.reactor: Any = None
        self.trace: list[tuple[Any, ...]] = []
        self.violations: list[str] = []
        self.ops: list[Op] = []
        self.chunks: list[tuple[float, bytes]] = []
        self.cursor = 0
        self.offset = 0
        self.consumed = 0
        self.emitted = 0
        self.eof_at: float | None = None
        self.exit_at: float | None = None
        self.exit_outcome: object = None
        self.writes = 0
        self.fault_at: int | None = None
        self.clock = Clock(self)
        self.gateway = Gateway(self)
        self.process = Process(self)
        self.acts = Acts(self)

    def mark(self) -> None:
        self.fault_at = len(self.trace)

    # ------------------------------------------------------------ peer stdout / exit

    def emit(self, at: float, data: bytes) -> None:
        index = bisect.bisect_right([when for when, _ in self.chunks], at)
        floor = self.cursor + 1 if self.offset else self.cursor  # never before consumed bytes
        self.chunks.insert(max(index, floor), (at, data))
        self.emitted += len(data)

    def say(self, at: float, events: list[dict[str, object]]) -> None:
        data = b"".join(ht.line_bytes(event) for event in events)
        if self.script.fragment and len(data) > 1:
            half = len(data) // 2
            self.emit(at, data[:half])
            self.emit(at + DT, data[half:])
        else:
            self.emit(at, data)

    def finish(
        self, at: float, code: int | None = None, *, eof: bool = True, outcome: object = None
    ) -> None:
        if eof:
            self.eof_at = at
        self.exit_at = at
        if outcome is None:
            outcome = self.b.WaitOutcome(status=self.r.ExitStatus.EXITED, code=code, reaped=True)
        self.exit_outcome = outcome

    def fail_later(self, at: float) -> None:
        """The controller's own deadline: ABORT + TERMINAL ABORTED, then natural exit 68."""
        self.say(at, [ht.abort(0, "CASE", "TIMEOUT", False), ht.terminal("ABORTED")])
        self.finish(at + 0.5, code=68)

    def readable_at(self) -> float | None:
        if self.cursor < len(self.chunks):
            return self.chunks[self.cursor][0]
        return self.eof_at

    def take(self, limit: object) -> bytes:
        size = limit if type(limit) is int else 0
        out = bytearray()
        while self.cursor < len(self.chunks) and len(out) < size:
            at, data = self.chunks[self.cursor]
            if at > self.t:
                break
            piece = data[self.offset : self.offset + size - len(out)]
            out += piece
            self.offset += len(piece)
            if self.offset == len(data):
                self.cursor += 1
                self.offset = 0
        self.consumed += len(out)
        return bytes(out)

    # ------------------------------------------------------------ port reactions

    def attach(self, role: str, slot: Any, op: Op) -> None:
        """The port contract: bind the inert op into the ORIGINAL slot before any effect."""
        fault = self.script.faults.pop(role, None)
        if fault == "no_bind":
            self.mark()
            return
        slot.bind(op)
        if fault == "double_bind":
            self.mark()
            slot.bind(Op(self, role + "-second"))
        if fault == "bind_raise":
            self.mark()
            raise RuntimeError("synthetic bind fault")
        if isinstance(fault, tuple):
            point, control = fault
            if point == "begin":
                self.mark()
                raise control
            op.fault = control
        elif fault in ("done_raise", "done_int", "outcome_raise") or isinstance(fault, Replace):
            op.fault = fault

    def on_spawn(self, begin: float) -> Any:
        if self.script.spawn is not None:
            return self.script.spawn(self, begin)
        ready = begin + DT
        self.emit(ready + DT, ht.line_bytes(ht.GUARD))
        return ready, self.b.SpawnOutcome(ok=True, process=self.process)

    def on_write(self, payload: bytes, begin: float, deadline: Any) -> Any:
        index = self.writes
        self.writes += 1
        plan = self.script.writes.get(index)
        if plan is not None:
            return plan(self, payload, begin, deadline)
        ready = begin + 3 * DT
        answer_at = begin + DT if self.script.early else ready + DT  # early: before completion
        self.say(answer_at, respond(payload, self.script.live))
        sent = self.b.WriteOutcome(delivery=self.r.Delivery.SENT, bytes_written=len(payload))
        return ready, sent

    def on_act(self, act: Any, begin: float, deadline: Any) -> Any:
        kind = act.kind.name
        plan = self.script.acts.get((kind, act.case))
        if plan is not None:
            return plan(self, act, begin, deadline)
        ready = begin + 2 * DT
        if kind == "KILL_GATEWAY":
            self.finish(begin + DT if self.script.exit_first else ready + DT, code=137)
        return ready, act_ok(self, kind)


def act_ok(world: World, kind: str, facts: object = None) -> Any:
    chosen = FACTS[kind](world.r) if facts is None else facts
    return world.b.ActOutcome(status=world.r.ActStatus.OK, facts=chosen)


def write_plan(mode: str, written: int = 0) -> Any:
    def plan(world: World, _payload: bytes, begin: float, deadline: Any) -> tuple[Any, object]:
        if mode == "LIVE":  # the peer never reads stdin: alive past the bridge deadline
            world.fail_later(deadline + 1.0)
            return None, None
        ready = begin + 3 * DT
        world.fail_later(ready + 0.5)
        delivery = getattr(world.r.Delivery, mode)
        return ready, world.b.WriteOutcome(delivery=delivery, bytes_written=written)

    return plan


def act_plan(mode: str) -> Any:
    def plan(world: World, _act: Any, begin: float, deadline: Any) -> tuple[Any, object]:
        if mode == "LIVE":
            world.fail_later(deadline + 1.0)
            return None, None
        ready = begin + 2 * DT
        world.fail_later(ready + 0.5)
        return ready, world.b.ActOutcome(status=getattr(world.r.ActStatus, mode), facts=None)

    return plan

class Lab:
    """One fresh reactor, one bridge, one fake world; status() purity checked on settle."""

    def __init__(
        self, core: types.ModuleType, script: Script | None = None, budget: Any = None
    ) -> None:
        self.b = bridge_module()
        self.r = reactor_mod()
        self.world = World(self.b, self.r, script if script is not None else Script())
        self.reactor = fresh(core, self.r)
        self.budget = budget if budget is not None else budget_of(self.b)
        w = self.world
        w.reactor = self.reactor
        w.overall = T0 + self.budget.overall_s
        self.bridge = self.b.DuplexBridge(
            self.r, self.reactor, w.clock, w.gateway, w.acts, self.budget
        )
        w.clock.bridge = self.bridge

    def run(self) -> Any:
        result = self.bridge.run()
        assert result.status == self.settle()
        return result

    def refuse(self, category: str, reactor_refusal: str | None = None) -> Any:
        with pytest.raises(self.b.BridgeRefusal) as caught:
            self.bridge.run()
        closed(self.b, caught.value, category, reactor_refusal)
        status = self.settle()
        assert status.fault is getattr(self.b.BridgeFault, category)
        assert status.reactor_refusal == reactor_refusal
        assert status.retained is True  # a closed error is never cleanup
        held = self.b.BridgePhase.REFUSED if reactor_refusal else self.b.BridgePhase.HELD
        assert status.phase is held
        return status

    def settle(self) -> Any:
        """Common invariants of every finished run; status() never calls a port/clock/op."""
        w, r = self.world, self.r
        assert w.violations == []
        before = len(w.trace)
        first, second = self.bridge.status(), self.bridge.status()
        assert len(w.trace) == before and first == second
        assert all(op.outcome_calls <= 1 for op in w.ops)
        assert len(events(w, "spawn")) <= 1 and len(events(w, "wait")) <= 1
        for _, deadline, _ in events(w, "spawn"):
            assert deadline == min(T0 + self.budget.spawn_s, w.overall)
        for _, max_bytes, deadline, _, consumed in events(w, "read"):
            assert max_bytes == min(READ_CHUNK, MAX_STDOUT - consumed + 1)
            assert deadline == w.overall
        for _, deadline, _ in events(w, "wait"):
            assert deadline == w.overall
        for _, payload, deadline, _, pending in events(w, "write"):
            assert type(pending) is r.Send and pending.payload == payload
            assert deadline == min(pending.deadline, w.overall)
        for _, _, _, deadline, _, act, pending in events(w, "act"):
            assert type(act) is r.Act and act is pending
            assert deadline == min(act.deadline, w.overall)
        if w.fault_at is not None:  # no hidden effect after the first port/control fault
            assert not any(event[0] in BEGINS for event in w.trace[w.fault_at :])
        return first


def closed(b: Any, error: BaseException, category: str, reactor_refusal: str | None = None) -> None:
    refusal: Any = error
    assert type(refusal) is b.BridgeRefusal
    assert refusal.args == (category,)
    assert refusal.category is getattr(b.BridgeFault, category)
    assert refusal.reactor_refusal == reactor_refusal
    assert refusal.__cause__ is None and refusal.__context__ is None
    text = str(refusal) + repr(refusal)
    assert all(secret not in text for secret in SECRET_TEXTS)


def events(world: World, kind: str) -> list[tuple[Any, ...]]:
    return [event for event in world.trace if event[0] == kind]


def written(world: World) -> list[bytes]:
    return [event[1] for event in events(world, "write")]


def acted(world: World) -> list[tuple[str, int]]:
    return [(event[1], event[2]) for event in events(world, "act")]


def ops_of(world: World, role: str) -> list[Op]:
    return [op for op in world.ops if op.role == role]


def view(status: Any, name: str) -> Any:
    assert tuple(item.name for item in status.slots) == SLOT_NAMES
    return status.slots[SLOT_NAMES.index(name)]


def verdicts_until(case: int) -> list[str]:
    return ["PASS"] * (case - 1) + ["FAIL"] + ["NOT_RUN"] * (25 - case)


def drained_during_every_write(world: World) -> bool:
    """Each write saw a read harvest before its own completion (no stdin/stdout lock)."""
    pending, seen, writes = False, False, 0
    for event in world.trace:
        if event[0] == "write":
            pending, seen, writes = True, False, writes + 1
        elif event[:2] == ("outcome", "read") and pending:
            seen = True
        elif event[:2] == ("outcome", "write"):
            if not seen:
                return False
            pending = False
    return writes == 51


class Trap:
    """Foreign value that fails if a repr/str/format or attribute inspection touches it."""

    def __init__(self, touched: list[str]) -> None:
        self.touched = touched

    def __repr__(self) -> str:
        self.touched.append("repr")
        raise AssertionError("foreign value inspected")

    def __str__(self) -> str:
        self.touched.append("str")
        raise AssertionError("foreign value inspected")

    def __format__(self, _spec: str) -> str:
        self.touched.append("format")
        raise AssertionError("foreign value inspected")

    def __getattr__(self, name: str) -> Any:
        self.touched.append(name)
        raise AssertionError("foreign value inspected")


class InertOp:
    """Op shape whose repr must never be invoked by the slot or its views."""

    def done(self) -> bool:
        return False

    def outcome(self) -> object:
        raise AssertionError("never harvested")

    def __repr__(self) -> str:
        raise AssertionError("op repr inspected")

REACTOR_VARIANTS = (
    "namespace",
    "missing-ActKind",
    "Wait-not-a-type",
    "subclass-type",
    "foreign-instance",
    "not-fresh",
    "refused",
)


@pytest.mark.parametrize("variant", REACTOR_VARIANTS)
def test_d1_reactor_binding_is_exact_and_fresh(core: types.ModuleType, variant: str) -> None:
    b, r = bridge_module(), reactor_mod()
    world = World(b, r, Script())
    reactor: Any = fresh(core, r)
    module: object = r
    if variant == "namespace":
        module = SimpleNamespace(**ht._reactor_namespace())
    elif variant == "missing-ActKind":
        module = reactor_variant(ActKind=ht.DROP)
    elif variant == "Wait-not-a-type":
        module = reactor_variant(Wait=7)
    elif variant == "subclass-type":  # the instance is not of the module's exact class

            reactor.feed(object())
    with pytest.raises(b.BridgeRefusal) as caught:
        b.DuplexBridge(module, reactor, world.clock, world.gateway, world.acts, budget_of(b))
    closed(b, caught.value, "REACTOR")
    assert world.trace == []


FRESH_TAMPERS = (
    "lines-bool",
    "lines-float",
    "controls-bool",
    "controls-float",
    "verdicts-list",
    "verdicts-one-pass",
    "verdicts-24",
)


def fresh_tamper(r: Any, case: str) -> dict[str, object]:
    verdicts = (r.Verdict.NOT_RUN,) * 25
    return {
        "lines-bool": {"lines": False},
        "lines-float": {"lines": 0.0},
        "controls-bool": {"controls_sent": False},
        "controls-float": {"controls_sent": 0.0},
        "verdicts-list": {"verdicts": list(verdicts)},
        "verdicts-one-pass": {"verdicts": (r.Verdict.PASS, *verdicts[1:])},
        "verdicts-24": {"verdicts": verdicts[1:]},
    }[case]


@pytest.mark.parametrize("case", FRESH_TAMPERS)
def test_d1_fresh_binding_needs_exact_counts_and_25_not_run(
    core: types.ModuleType, monkeypatch: pytest.MonkeyPatch, case: str
) -> None:
    """R8.6 on the REAL instance: only its public status view is altered, then restored."""
    b, r = bridge_module(), reactor_mod()
    world = World(b, r, Script())
    reactor = fresh(core, r)
    genuine = r.HostReactor.status
    changes = fresh_tamper(r, case)  # equal-by-== values that are NOT the exact fresh binding

    def altered(self: Any) -> Any:
        return dataclasses.replace(genuine(self), **changes)

    monkeypatch.setattr(r.HostReactor, "status", altered)
    with pytest.raises(b.BridgeRefusal) as caught:
        b.DuplexBridge(r, reactor, world.clock, world.gateway, world.acts, budget_of(b))
    closed(b, caught.value, "REACTOR")
    assert world.trace == []

def test_d1_attempt_is_consumed_before_the_first_clock_callback(core: types.ModuleType) -> None:
    lab = Lab(core)
    seen: list[object] = []

    def first_now(clock: Clock) -> object:
        if not seen:
            status = clock.bridge.status()  # pure read from inside the callback
            seen.append((status.attempt_consumed, status.phase.name))
            with pytest.raises(lab.b.BridgeRefusal) as inner:
                clock.bridge.run()
            seen.append(inner.value.args)
        return clock.world.t

    lab.world.clock.on_now = first_now
    result = lab.run()
    assert seen == [(True, "STARTING"), ("REENTRY",)]
    assert result.done.outcome.name == "COMPLETE_CANDIDATE"


def test_d1_run_is_one_shot_and_the_first_fault_is_sticky(core: types.ModuleType) -> None:
    lab = Lab(core)
    first = lab.run()
    count = len(lab.world.trace)
    with pytest.raises(lab.b.BridgeRefusal) as again:
        lab.bridge.run()
    closed(lab.b, again.value, "REENTRY")
    after = lab.bridge.status()
    assert len(lab.world.trace) == count
    assert after.slots == first.status.slots
    assert after.reactor_status == first.status.reactor_status
    assert first.done.outcome.name == "COMPLETE_CANDIDATE"

    other = Lab(core, Script(faults={"read": "done_raise"}))
    other.refuse("PORT")
    count = len(other.world.trace)
    with pytest.raises(other.b.BridgeRefusal) as reentry:
        other.bridge.run()
    closed(other.b, reentry.value, "REENTRY")
    assert len(other.world.trace) == count
    assert other.bridge.status().fault is other.b.BridgeFault.PORT  # the first fault stays


def test_d1_operation_slot_binds_once_and_offers_no_clear() -> None:
    b = bridge_module()
    slot = b.OperationSlot("read")
    assert (slot.name, slot.op, slot.occupied) == ("read", None, False)
    public = sorted(name for name in dir(slot) if not name.startswith("_"))
    assert public == ["bind", "name", "occupied", "op"]
    shapeless = (
        object(),
        SimpleNamespace(done=True, outcome=InertOp().outcome),
        SimpleNamespace(done=InertOp().done),
    )
    for bad in shapeless:
        with pytest.raises(b.BridgeRefusal) as caught:
            slot.bind(bad)
        closed(b, caught.value, "PORT")
        assert (slot.op, slot.occupied) == (None, False)
    first = InertOp()
    slot.bind(first)
    with pytest.raises(b.BridgeRefusal) as occupied:
        slot.bind(InertOp())
    closed(b, occupied.value, "PORT")
    assert slot.op is first and slot.occupied is True  # never replaced
    assert isinstance(repr(slot), str)  # never invokes the op repr


FAULT_POINTS = ("no_bind", "bind_raise", "double_bind", "done_raise", "done_int", "outcome_raise")


def exit_with_guard(code: int) -> Any:
    def spawn(world: World, begin: float) -> tuple[float | None, object]:
        ready = begin + DT
        world.emit(ready + DT, ht.line_bytes(ht.GUARD))
        world.finish(ready + DT, code=code)
        return ready, world.b.SpawnOutcome(ok=True, process=world.process)

    return spawn


@pytest.mark.parametrize("point", FAULT_POINTS)
@pytest.mark.parametrize("role", SLOT_NAMES)
def test_d1_fault_at_each_port_boundary_retains_the_original(
    core: types.ModuleType, role: str, point: str
) -> None:
    script = Script(faults={role: point})
    if role == "wait":
        script.spawn = exit_with_guard(68)
    lab = Lab(core, script)
    status = lab.refuse("PORT")
    w = lab.world
    item, original = view(status, role), ops_of(w, role)[0]
    if point == "no_bind":
        assert (item.occupied, item.op) == (False, None)
    else:
        assert item.occupied is True and item.op is original
    assert status.process is (None if role == "spawn" else w.process)
    pending = status.reactor_status.pending
    if role == "write":
        assert pending.purpose.name == "INIT" and status.reactor_status.controls_sent == 0
    if role == "act":
        assert (pending.kind.name, pending.case) == ("TERMINATE_ONE_BACKEND", 20)


SPAWN_CASES = (
    "live",
    "failed-with-process",
    "failed-without-process",
    "ok-shapeless-process",
    "ok-text",
    "not-an-outcome",
)


@pytest.mark.parametrize("case", SPAWN_CASES)
def test_d1_spawn_outcome_adopts_the_process_before_refusing(
    core: types.ModuleType, case: str
) -> None:
    shapeless = object()

    def spawn(world: World, begin: float) -> tuple[float | None, object]:
        b, proc = world.b, world.process
        if case == "live":
            return None, None
        made: object = {
            "failed-with-process": lambda: b.SpawnOutcome(ok=False, process=proc),
            "failed-without-process": lambda: b.SpawnOutcome(ok=False, process=None),
            "ok-shapeless-process": lambda: b.SpawnOutcome(ok=True, process=shapeless),
            "ok-text": lambda: b.SpawnOutcome(ok="yes", process=proc),
            "not-an-outcome": object,
        }[case]()
        return begin + DT, made

    lab = Lab(core, Script(spawn=spawn))
    spawn_fault = case in ("live", "failed-with-process", "failed-without-process")
    status = lab.refuse("SPAWN" if spawn_fault else "PORT")
    w = lab.world
    adopted = {  # adopted from the cached outcome BEFORE ok/shape validation
        "failed-with-process": w.process,
        "ok-shapeless-process": shapeless,
        "ok-text": w.process,
    }
    assert status.process is adopted.get(case)
    item, op = view(status, "spawn"), ops_of(w, "spawn")[0]
    assert item.occupied is True and item.op is op and item.done is (case != "live")
    assert item.outcome is (None if case == "live" else op.result)
    assert events(w, "read") == [] and events(w, "wait") == []
    if case == "live":
        assert w.t == T0 + SPAWN

    for text in (repr(result), repr(status), str(status)):
        assert all(secret not in text for secret in SECRET_TEXTS)

BAD_WRITES: dict[str, tuple[str, object]] = {  # (delivery spec, bytes_written) for GO1
    "not-sent-with-bytes": ("NOT_SENT", 3),
    "sent-short": ("SENT", 15),
    "sent-bool-count": ("SENT", True),
    "unknown-over": ("UNKNOWN", 17),
    "unknown-negative": ("UNKNOWN", -1),
    "text-delivery": ("text", 16),
    "foreign-enum": ("foreign", 16),
    "not-an-outcome": ("object", 0),
}


def bad_write(world: World, spec: str, count: object) -> object:
    if spec == "object":
        return object()
    if spec == "text":
        delivery: object = "SENT"
    elif spec == "foreign":
        delivery = world.r.ActStatus.OK
    else:
        delivery = getattr(world.r.Delivery, spec)
    return world.b.WriteOutcome(delivery=delivery, bytes_written=count)


@pytest.mark.parametrize("case", sorted(BAD_WRITES))
def test_d3_malformed_write_outcome_is_port_and_retained(core: types.ModuleType, case: str) -> None:
    def plan(world: World, _payload: bytes, begin: float, _deadline: Any) -> tuple[Any, object]:

def bad_act(world: World, spec: str, facts_spec: str) -> object:

def init_not_sent(lines: list[tuple[float, dict[str, object]]], alive_at: float | None) -> Any:
    def plan(world: World, _payload: bytes, begin: float, _deadline: Any) -> tuple[Any, object]:
        ready = begin + 3 * DT
        for offset, event in lines:
            world.say(ready + offset, [event])
        if alive_at is not None:  # the port positively reports a live process
            alive = world.b.WaitOutcome(status=world.r.ExitStatus.NO_EXIT, code=None, reaped=False)
            world.finish(ready + alive_at, eof=False, outcome=alive)
        return ready, world.b.WriteOutcome(delivery=world.r.Delivery.NOT_SENT, bytes_written=0)

    return plan

def interrupt_on(role: str, control: BaseException) -> Any:
    def hook(clock: Clock, watched: Any, _deadline: Any) -> object:
        if any(getattr(slot, "name", None) == role for slot in watched):
            clock.world.mark()
            raise control
        return None

    return hook


CONTROL_POINTS = ("poll-during-write", "begin-act-after-bind", "spawn-outcome")


@pytest.mark.parametrize("where", CONTROL_POINTS)
def test_d6_first_control_is_relaunched_intact_without_new_callbacks(
    core: types.ModuleType, where: str
) -> None:
    control: BaseException = KeyboardInterrupt()
    if where == "begin-act-after-bind":
        control = SystemExit(3)
    script = Script()
    if where == "begin-act-after-bind":
        script.faults["act"] = ("begin", control)
    elif where == "spawn-outcome":
        script.faults["spawn"] = ("outcome", control)
    lab = Lab(core, script)
    if where == "poll-during-write":
        lab.world.clock.on_poll = interrupt_on("write", control)
    with pytest.raises(type(control)) as caught:
        lab.bridge.run()
    w, b = lab.world, lab.b
    assert caught.value is control  # the same instance, never a replacement
    assert len(w.trace) == w.fault_at  # no done/outcome/clock/port callback in the handler
    status = lab.settle()
    assert (status.fault, status.phase, status.retained) == (
        b.BridgeFault.CONTROL,
        b.BridgePhase.HELD,
        True,
    )
    role = {"poll-during-write": "write", "begin-act-after-bind": "act"}.get(where, "spawn")
    item = view(status, role)
    assert item.occupied is True and item.op is ops_of(w, role)[0]
    assert status.process is (None if role == "spawn" else w.process)
    with pytest.raises(b.BridgeRefusal) as again:
        lab.bridge.run()
    closed(b, again.value, "REENTRY")
    assert len(w.trace) == w.fault_at and lab.bridge.status().fault is b.BridgeFault.CONTROL


CLEANUP_WORDS = ("quiesce", "ticket", "nonce", "released", "cleanup", "absent")
CLEANUP_ATTRS = frozenset(
    {"close", "kill", "terminate", "cancel", "communicate", "wait_for", "send_signal"}
)
SOURCE_NAMES = (
    "w1_bootstrap_channel",
    "oracle9_w1_run",
    "host_reactor",
    "driver",
    "oracle9_w1_main",
    "oracle9_w1_controller",
)


def test_d6_module_is_pure_and_offers_no_cleanup_api() -> None:
    tree = ast.parse(BRIDGE_FILE.read_text(encoding="utf-8"))
    identifiers: list[str] = []
    for node in ast.walk(tree):
        imported: list[str] = []
        if isinstance(node, ast.Import):
            imported = [alias.name for alias in node.names]
        elif isinstance(node, ast.ImportFrom):
            imported = [node.module or ""]
            identifiers += [alias.name for alias in node.names]
        for name in imported:
            assert name.split(".")[0] not in ht.FORBIDDEN_IMPORTS
            assert not any(part in name for part in SOURCE_NAMES)
        if isinstance(node, ast.Call) and isinstance(node.func, ast.Name):
            assert node.func.id not in ht.FORBIDDEN_CALLS | {"compile"}
        if isinstance(node, ast.Attribute):
            assert node.attr not in ht.CLOCK_ATTRS | CLEANUP_ATTRS
            identifiers.append(node.attr)
        elif isinstance(node, ast.Name):
            identifiers.append(node.id)
        elif isinstance(node, ast.FunctionDef | ast.ClassDef):
            identifiers.append(node.name)
        elif isinstance(node, ast.arg):
            identifiers.append(node.arg)
    flagged = [name for name in identifiers if any(word in name.lower() for word in CLEANUP_WORDS)]
    assert flagged == []
