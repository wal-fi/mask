# -- simulated handle-based filesystem --


class Node:
    def __init__(self, name: str, kind: str, data: bytes = b"") -> None:
        self.name, self.kind, self.data = name, kind, data
        self.children: dict[str, Node] = {}
        self.file_id = node_id()
        self.overrides: dict[str, Any] = {}

    def info(self, glob: Any) -> Any:
        fields: dict[str, Any] = {
            "reparse": False,
            "directory": self.kind == "DIR",
            "delete_pending": False,
            "volume": VOLUME,
            "file_id": self.file_id,
            "links": 1,
            "end_of_file": len(self.data) if self.kind == "FILE" else 0,
            "change_time": 1000,
            "streams": ("::$DATA",) if self.kind == "FILE" else (),
        }
        fields.update(self.overrides)
        return glob["Info"](**fields)


class FakeBackend:
    """Exactly the six fixed operations; every call is recorded."""

    def __init__(self, glob: Any, root: Node) -> None:
        self.glob, self.root = glob, root
        self.handles: dict[int, Node] = {}
        self.token = 0x10000
        self.ops: list[tuple[str, str]] = []
        self.counts: dict[tuple[str, str], int] = {}
        self.hooks: dict[Any, Any] = {}
        self.acquired: list[int] = []
        self.closed: list[int] = []

    def _call(self, op: str, name: str, default: Any, extra: Any) -> Any:
        self.ops.append((op, name))
        key = (op, name)
        count = self.counts[key] = self.counts.get(key, 0) + 1
        hook = self.hooks.get((op, name, count), self.hooks.get(key))
        return default() if hook is None else hook(default, extra)

    def _fill(self, slot: Any, node: Node) -> None:
        self.token += 1
        self.handles[self.token] = node
        self.acquired.append(self.token)
        slot.value = self.glob["Handle"](self.token)

    def _node(self, handle: Any) -> Node:
        return self.handles[handle.token]

    def open_anchor(self, slot: Any) -> Any:
        extra = SimpleNamespace(slot=slot, node=self.root, backend=self)
        return self._call("open", "<anchor>", lambda: self._fill(slot, self.root), extra)

    def open_child(self, parent: Any, name: str, kind: str, slot: Any) -> Any:
        node = self._node(parent).children.get(name)

        def default() -> None:
            if node is None or node.kind != kind:
                raise FileNotFoundError("fictitious-missing")
            self._fill(slot, node)

        extra = SimpleNamespace(slot=slot, node=node, backend=self)
        return self._call("open", name, default, extra)

    def query(self, handle: Any) -> Any:
        node = self._node(handle)
        extra = SimpleNamespace(node=node, backend=self)
        return self._call("query", node.name, lambda: node.info(self.glob), extra)

    def list_dir(self, handle: Any, max_entries: int) -> Any:
        assert type(max_entries) is int and max_entries == 16  # exact policy bound
        node = self._node(handle)
        entry = self.glob["Entry"]

        def default() -> Any:
            return tuple(
                entry(name=name, directory=child.kind == "DIR", file_id=child.file_id)
                for name, child in sorted(node.children.items())
            )

        return self._call("list", node.name, default, SimpleNamespace(node=node, backend=self))

    def read(self, handle: Any, limit: int) -> Any:
        node = self._node(handle)
        extra = SimpleNamespace(node=node, backend=self)
        return self._call("read", node.name, lambda: node.data[:limit], extra)

    def close(self, handle: Any) -> Any:
        self.closed.append(handle.token)
        node = self.handles.get(handle.token)
        name = node.name if node is not None else "<unknown>"
        return self._call("close", name, lambda: True, SimpleNamespace(node=node, backend=self))


def path(parent: Node, *names: str) -> Node:
    node = parent
    for name in names:
        node = node.children.setdefault(name, Node(name, "DIR"))
    return node


def make_tree(bundle: Any, source: bytes) -> Node:
    root = Node("<anchor>", "DIR")
    poc = path(root, *REPO_CHAIN)
    poc.children[VERIFY_NAME] = Node(VERIFY_NAME, "FILE", source)
    public = path(poc, "evidence", PUBLIC_NAME)
    public.children["summary.json"] = Node("summary.json", "FILE", bundle.summary_raw)
    public.children["transport.json"] = Node("transport.json", "FILE", bundle.transport_raw)
    private = path(root, *TEMP_CHAIN, PRIVATE_NAME)
    for name, raw in bundle.private.items():
        case, leaf = name.split("/")
        path(private, case).children[leaf] = Node(leaf, "FILE", raw)
    return root


def named(root: Node, name: str) -> Node:
    pending = [root]
    while pending:
        node = pending.pop(0)
        if node.name == name:
            return node
        pending.extend(child for _key, child in sorted(node.children.items()))
    raise AssertionError("fixture node not found")

def prepare(lab: Any, scenario: str = "approved", source: Any = None) -> Any:
    builder = lab.builder
    bundle = builder["build"](builder[scenario]())
    root = make_tree(bundle, lab.source if source is None else source)
    backend = FakeBackend(lab.glob, root)
    lab.monkeypatch.setitem(lab.glob, "_backend", lambda: backend)
    return SimpleNamespace(bundle=bundle, root=root, backend=backend)


def collect(lab: Any) -> Any:
    return lab.glob["collect"](NONCE, ARGV0)


def invalid(lab: Any, reason: str) -> None:
    assert collect(lab) == ("INVALID", reason, None, None)


def closed_once(backend: FakeBackend) -> bool:
    """Every acquired token closed, and none closed twice."""
    unique = len(set(backend.closed)) == len(backend.closed)
    return unique and sorted(backend.closed) == sorted(backend.acquired)


def hook_raise(error: BaseException) -> Any:
    def hook(_default: Any, _extra: Any) -> Any:
        raise error

    return hook


def hook_value(value: Any) -> Any:
    def hook(_default: Any, _extra: Any) -> Any:
        return value

    return hook


def hook_map(function: Any) -> Any:
    def hook(default: Any, _extra: Any) -> Any:
        return function(default())

    return hook


def fill_then(error: BaseException) -> Any:
    def hook(default: Any, _extra: Any) -> Any:
        default()
        raise error

    return hook

def test_module_source_is_pure() -> None:
    tree = ast.parse(MODULE.read_bytes())
    imported = {
        alias.name.split(".")[0]
        for node in ast.walk(tree)
        if isinstance(node, ast.Import)
        for alias in node.names
    }
    imported |= {
        (node.module or "").split(".")[0]
        for node in ast.walk(tree)
        if isinstance(node, ast.ImportFrom)
    }
    assert imported <= {"dataclasses", "hashlib", "re"}
    names = [node.id for node in ast.walk(tree) if isinstance(node, ast.Name)]
    assert not set(names) & {"open", "print", "input", "eval", "__import__", "globals"}
    assert names.count("exec") <= 1
    assert not any(
        isinstance(node, ast.If) and "__main__" in ast.unparse(node.test) for node in ast.walk(tree)
    )

def _flip_entry_kind(target: str) -> Any:
    def hook(default: Any, _extra: Any) -> Any:
        result = []
        for entry in default():
            item = entry
            if entry.name == target:
                item = dataclasses.replace(entry, directory=not entry.directory)
            result.append(item)
        return tuple(result)

    return hook

def _swap_id(target: str, keep_first_half: bool) -> Any:
    def hook(default: Any, _extra: Any) -> Any:
        result = []
        for entry in default():
            item = entry
            if entry.name == target:
                fid = entry.file_id
                fid = fid[:8] + bytes(8) if keep_first_half else bytes(8) + fid[8:]
                item = dataclasses.replace(entry, file_id=fid)
            result.append(item)
        return tuple(result)

    return hook

def _bad_info(glob: Any, kind: str) -> Any:
    good = Node("summary.json", "FILE", b"x").info(glob)
    fields = {name: getattr(good, name) for name in INFO_FIELDS}
    if kind == "object":
        return object()
    if kind == "subclass":
        return type("Sub", (glob["Info"],), {})(**fields)
    changes = {
        "volume-bool": {"volume": True},
        "volume-negative": {"volume": -1},
        "volume-2**64": {"volume": 1 << 64},
        "id-15": {"file_id": bytes(15)},
        "id-bytearray": {"file_id": bytearray(16)},
        "links-0": {"links": 0},
        "links-2**32": {"links": 1 << 32},
        "eof-negative": {"end_of_file": -1},
        "eof-2**63": {"end_of_file": 1 << 63},
        "time-high": {"change_time": 1 << 63},
        "time-low": {"change_time": -(1 << 63) - 1},
        "streams-list": {"streams": ["::$DATA"]},
        "streams-9": {"streams": ("::$DATA",) * 9},
        "streams-empty-name": {"streams": ("",)},
        "streams-nul": {"streams": ("a\x00",)},
        "streams-subclass": {"streams": (SneakyStr("::$DATA"),)},
        "reparse-int": {"reparse": 0},
    }[kind]
    return glob["Info"](**{**fields, **changes})

    extra.slot.value = extra.backend.glob["Handle"](extra.backend.acquired[0])


def test_alias_is_quarantined_and_poisons(lab: Any) -> None:

    invalid(lab, "COLLECT_CUSTODY")

    assert lab.glob["_POISON"] != []
    again = prepare(lab)
    invalid(lab, "COLLECT_POISONED")
    assert again.backend.ops == []
