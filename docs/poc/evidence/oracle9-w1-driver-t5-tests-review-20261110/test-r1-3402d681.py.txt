"""T5 EOF probe (case 24) over deterministic fake ports; RED-first, no physical effect.

Contract for the FUTURE amendment of ``deploy/oracle9_w1_driver.py`` described by
T5-DESIGN v3 (5eb14030) and T5-DESIGN-REVIEW. The current driver (39a68904) has eight
ports and emits T5_PENDING, so every feature case here is expected to fail first with
the new ``Ports(children=...)`` constructor; that API RED alone proves no temporal or
ownership behavior. The driver is loaded lazily (``d1.driver()``) inside the feature
tests, never at collection. The ``test_sanity_*`` cases do not load the driver.

Reused, unchanged: the D1 deterministic world (Link, Factory/FakeSession over the REAL
BudgetLedger, Pair/Threads, PureReader, Guard) from ``tests/test_oracle9_w1_driver.py``.
Added here only: a ``children`` port fake (socket ends, explicit stdin pipe ends and a
child process), ``ids.nonce()``, a clock able to return one invalid reading, and a
delegating recording subclass of the REAL public ``BudgetLedger`` installed only in
the loaded driver namespace (it never invents a ledger result; at most it raises a
control interruption AFTER the real call). Old helpers/assertions are not modified;
case-24 RESULT and SUMMARY use a local exact checker because the old one rejects the
new EOF schema by design.

Prospective API fixed by this file (names from T5-DESIGN v3 §3/§4):

- ``Ports(..., children=...)`` (9th field) and ``ids.nonce() -> bytes`` (exactly 32).
- ``children.pair() -> (parent, child)``; ``parent.recv_exact(n, *, deadline)``,
  ``parent.close() -> bool``; ``child.fd``, ``child.close() -> bool``.
- ``children.stdin() -> (reader, writer)``; ``reader.fd``, ``reader.close() -> bool``;
  ``writer.write_all(packet, *, deadline) -> bool``, ``writer.close() -> bool``.
- ``children.spawn(child, reader, *, parent_pid, address_bytes, idle_ms) -> proc``;
  ``proc.pid``, ``proc.wait(*, deadline) -> int | None``, ``proc.kill() -> bool``.
- Owner unchanged: token in ``session_token`` (set BEFORE adopt), the four ends in
  ``endpoints``, the process in ``handles``, uncertain creations in ``unknown`` as
  ``probe_pair``/``probe_stdin``/``probe_process``.
- RESULT case 24: ``kind`` "EOF", ``verdict`` PASS/FAIL, ``reason`` null or one of
  EXIT_NONZERO/EXIT_LATE/WAIT_LATE/NO_EXIT, ``release`` "RELEASED", other fields null.
  SUMMARY no longer always contains 24; ``global_pass`` stays false.

D5-12 uses two declared deterministic LOGICAL fault injections inside the recording
subclass, for the probe token only, each produced by REAL public ledger algorithms:
``adopt_refuse`` calls the real ``begin_drain`` right before the real
``adopt_session`` (a genuine STATE refusal, then the driver's real ``retain`` gives
HELD); ``release_held`` takes a real output pin (``begin_output``, 1 + 1 bytes) right
after the real adoption and keeps it strongly referenced in the world, so the
driver's real ``release`` genuinely returns HELD. They model ledger state drift,
not concurrency, a physical receiver/output allocation or native evidence.
Fakes model contracts only: kernel EOF, fd inheritance/close-on-exec, waitpid and
returncode, the installed worker env/guard, a common monotonic clock and PostgreSQL
backend closure remain native Oracle proof. Nothing here is native evidence.
"""

def eof_event(verdict: str, reason: str | None) -> dict[str, Any]:
    return {
        "bytes": None,
        "case": T5,
        "e": "RESULT",
        "kind": "EOF",
        "next_seq": 49,
        "reason": reason,
        "release": "RELEASED",
        "rows": None,
        "sink": None,
        "stop_applied": None,
        "truncated": None,
        "verdict": verdict,
    }


EOF_PASS = eof_event("PASS", None)


# ---------------------------------------------------------------- local exact checkers


def check_eof(event: dict[str, Any]) -> bool:
    verdict, reason = event["verdict"], event["reason"]
    coherent = (verdict == "PASS" and reason is None) or (
        verdict == "FAIL" and d1.exact_text(reason, FAIL_REASONS)
    )
    return (
        d1.exact_int(event["case"], T5, T5)
        and d1.exact_int(event["next_seq"], 49, 49)
        and d1.exact_text(verdict, ("PASS", "FAIL"))
        and d1.exact_text(event["kind"], ("EOF",))
        and coherent
        and d1.exact_text(event["release"], ("RELEASED",))
        and all(event[name] is None for name in ("rows", "truncated", "bytes", "sink"))
        and event["stop_applied"] is None
    )


def check_summary(event: dict[str, Any]) -> bool:
    failed = event["failed"]
    return (
        type(failed) is list
        and all(d1.exact_int(case, 1, 24) for case in failed)
        and failed == sorted(set(failed))
        and event["global_pass"] is False
        and d1.exact_int(event["prefix_cases"], 24, 24)
    )


def events_t5(lines: list[bytes]) -> list[dict[str, Any]]:
    """Same per-line rules as D1; only case-24 RESULT and SUMMARY use the new schema."""
    assert len(lines) <= 56
    result: list[dict[str, Any]] = []
    for line in lines:
        assert type(line) is bytes
        assert 0 < len(line) <= 512
        assert b"\n" not in line
        for leak in LEAKS:
            assert leak.encode() not in line
        event = json.loads(line)
        assert d1.canonical(event) == line
        assert type(event) is dict
        assert d1.exact_text(event.get("e"), tuple(d1.KEYS))
        assert set(event) == d1.KEYS[event["e"]]
        if event["e"] == "RESULT" and event["case"] == T5:
            assert check_eof(event)
        elif event["e"] == "SUMMARY":
            assert check_summary(event)
        else:
            assert d1.CHECKS[event["e"]](event)
        result.append(event)
    return result

_NO_READING = object()


class T5Clock(d1.Clock):
    """D1 clock plus one scripted invalid reading (NaN, inf, bool, object...)."""

    def __init__(self) -> None:
        super().__init__()
        self.bad: object = _NO_READING

    def monotonic(self) -> Any:
        if self.bad is not _NO_READING:
            value, self.bad = self.bad, _NO_READING
            return value
        return self.now


class T5Ids(d1.Ids):
    def __init__(self, world: T5World) -> None:
        super().__init__(world)
        self.t5world = world

    def session_id(self) -> Any:
        self.count += 1
        self.world.log.append(("id", self.count))
        script = self.t5world.t5
        if self.count == 6 and "t5_id" in script:
            return script["t5_id"]
        return bytes([self.count]) * 16

    def nonce(self) -> Any:
        world = self.t5world
        world.step("nonce")
        value = world.t5.get("nonce", NONCE)
        world.after("nonce")
        return value

class ParentEnd:
    """Trusted fake: exposes no send/write/shutdown, by contract of the binding."""

    def __init__(self, world: T5World) -> None:
        self.world = world
        self.inbox = bytearray()
        self.reads = 0

    def recv_exact(self, n: int, *, deadline: float) -> bytes:
        world = self.world
        names = ("guard_header", "guard_body", "opened_header", "opened_body")
        name = names[self.reads] if self.reads < len(names) else "extra_recv"
        self.reads += 1
        world.step(name, n=n, deadline=deadline)
        world.clock.now += world.t5.get("recv_delay", {}).get(name, 0.0)
        if len(self.inbox) < n:
            world.after(name)
            raise TimeoutError
        data = bytes(self.inbox[:n])
        del self.inbox[:n]
        world.after(name)
        return data

    def close(self) -> bool:
        return self.world.closing("parent_close")


class StdinWriter:
    def __init__(self, world: T5World) -> None:
        self.world = world
        self.wrote_ok = False

    def write_all(self, packet: bytes, *, deadline: float) -> bool:
        world = self.world
        world.t_write = world.clock.now
        world.step("write_all", deadline=deadline)
        world.packets.append(packet)
        world.clock.now += world.t5.get("write_delay", 0.0)
        self.wrote_ok = world.t5.get("write_ok", True) is True
        world.after("write_all")
        return self.wrote_ok

    def close(self) -> bool:
        world = self.world
        closed = world.closing("writer_close")
        if closed and self.wrote_ok and world.parent is not None and not world.opened_sent:
            world.opened_sent = True
            world.parent.inbox += world.opened_frame()
        return closed


class Proc:
    pid = PROC_PID

    def __init__(self, world: T5World) -> None:
        self.world = world
        self.killed = False

    def wait(self, *, deadline: float) -> Any:
        world = self.world
        world.step("wait", deadline=deadline)
        script = world.t5
        if not self.killed:
            failure = script.get("wait_raise")
            if failure is not None:
                raise failure
            rc = script.get("rc", 0)
            if rc is None:
                world.clock.now = max(world.clock.now, deadline)
                return None
            world.clock.now = max(world.clock.now, world.t_write + script.get("wait_at", 1.0))
            return rc
        if script.get("rc2_raise"):
            raise OSError("WAIT_UNCERTAIN")
        world.clock.now += script.get("rc2_delay", 0.1)
        return script.get("rc2", 0)

    def kill(self) -> bool:
        self.world.step("kill")
        self.killed = True
        return True


class ChildPort:
    def __init__(self, world: T5World) -> None:
        self.world = world

    def pair(self) -> tuple[ParentEnd, ChildEnd]:
        world = self.world
        world.t_pair = world.clock.now
        world.step("pair")
        if "pair_raise" in world.t5:
            raise world.t5["pair_raise"]
        world.parent, world.child = ParentEnd(world), ChildEnd(world)
        world.after("pair")
        return world.parent, world.child

    def stdin(self) -> tuple[StdinReader, StdinWriter]:
        world = self.world
        world.step("stdin")
        if "stdin_raise" in world.t5:
            raise world.t5["stdin_raise"]
        world.reader, world.writer = StdinReader(world), StdinWriter(world)
        world.after("stdin")
        return world.reader, world.writer

    def spawn(self, child: Any, reader: Any, **kwargs: Any) -> Proc:
        world = self.world
        world.step("spawn")
        world.spawn_args = (child, reader, kwargs)
        if "spawn_raise" in world.t5:
            raise world.t5["spawn_raise"]
        world.proc = Proc(world)
        assert world.parent is not None
        first = world.t5.get("first_frame")
        if first is None:
            first = wire.encode_guard(*world.t5.get("guard", (PROC_PID, 1, WORKER_AS)))
        world.parent.inbox += first
        world.after("spawn")
        return world.proc


def recording_ledger(world: T5World) -> type[BudgetLedger]:
    """Delegating subclass of the REAL public ledger; it records, never invents."""

    class Recording(BudgetLedger):
        def reserve_session(self, *args: Any, **kwargs: Any) -> Any:
            token = super().reserve_session(*args, **kwargs)
            world.reserves.append((token, kwargs))
            if len(world.reserves) == 6:
                world.t5_token = token
                world.step("reserve")
            return token

        def adopt_session(self, *args: Any, **kwargs: Any) -> Any:
            token = (*args, *kwargs.values())[0]
            probe = token is world.t5_token
            if probe:
                owner = world.owner
                world.adopt_owned = owner is not None and owner.session_token is token
                world.step("adopt")
                if world.t5.get("adopt_refuse"):  # declared drift: REAL begin_drain first
                    super().begin_drain(token)
            outcome = super().adopt_session(*args, **kwargs)
            if probe and world.t5.get("release_held"):  # declared drift: REAL output pin
                world.pin = super().begin_output(token, frame_bytes=1, overhead_bytes=1)
            if probe and world.t5.get("adopt_interrupt"):
                raise KeyboardInterrupt  # control AFTER the real adoption
            return outcome

        def retain(self, *args: Any, **kwargs: Any) -> Any:
            token = (*args, *kwargs.values())[0]
            outcome = super().retain(*args, **kwargs)
            if token is world.t5_token:
                world.step("retain")
            return outcome

        def release(self, *args: Any, **kwargs: Any) -> Any:
            token, confirmation = (*args, *kwargs.values())[:2]
            outcome = super().release(*args, **kwargs)
            if token is world.t5_token:
                world.t5_releases.append((confirmation, outcome))
                world.step("release")
            return outcome

    return Recording


class T5World(d1.World):
    clock: T5Clock  # narrowed: the D1 clock subtype with one scriptable invalid reading

    def __init__(self, **script: Any) -> None:
        super().__init__(**script)
        self.clock = T5Clock()
        self.pin: Any = None
        self.t5: dict[str, Any] = script.get("t5", {})
        self.owner: Any = None
        self.reserves: list[tuple[Any, dict[str, Any]]] = []
        self.t5_token: Any = None
        self.adopt_owned: bool | None = None
        self.t5_releases: list[tuple[Any, Any]] = []
        self.packets: list[bytes] = []
        self.snaps: dict[str, dict[str, Any]] = {}
        self.calls: list[tuple[str, dict[str, Any]]] = []
        self.parent: ParentEnd | None = None
        self.child: ChildEnd | None = None
        self.reader: StdinReader | None = None
        self.writer: StdinWriter | None = None
        self.proc: Proc | None = None
        self.spawn_args: Any = None
        self.opened_sent = False
        self.t_pair = 0.0
        self.t_write = 0.0

    # ------------------------------------------------------------ bookkeeping

    def snap(self) -> dict[str, Any]:
        owner = self.owner
        return {
            "endpoints": {id(item) for item in owner.endpoints},
            "handles": {id(item) for item in owner.handles},
            "unknown": tuple(owner.unknown),
            "token": owner.session_token,
        }

    def step(self, name: str, **details: Any) -> None:
        self.log.append(("t5", name))
        self.calls.append((name, details))
        if name not in self.snaps and self.owner is not None:
            self.snaps[name] = self.snap()

    def after(self, name: str) -> None:
        if name in self.t5.get("late", ()):
            self.clock.now += OPEN_CAP_S + 1.0  # returned only after the opening deadline
        bad = self.t5.get("bad_clock")
        if bad is not None and bad[0] == name:
            self.clock.bad = bad[1]

    def closing(self, name: str) -> bool:
        self.step(name)
        outcome = self.t5.get("close", {}).get(name, True)
        if isinstance(outcome, BaseException):
            raise outcome
        self.after(name)
        return outcome is True

    def opened_frame(self) -> bytes:
        packet = self.packets[-1]
        init = wire.decode_init(packet[HB:])
        mode = self.t5.get("opened", "ok")
        if mode == "error":
            failure = wire.ErrorReply(init.key, 0, wire.WireFailure.CONNECTION, False)
            return wire.encode_error(failure)
        tag = hmac.digest(init.nonce, init.snapshot, "sha256")
        if mode == "bad_hmac":
            tag = bytes([tag[0] ^ 1]) + tag[1:]
        return wire.encode_opened(tag)

    # ------------------------------------------------------------ views

    def trace(self) -> list[str]:
        result: list[str] = []
        for entry in self.log:
            if entry[0] == "t5":
                result.append(entry[1])
            elif entry[0] == "write" and entry[2] == T5 and entry[1] in ("READY", "RESULT"):
                result.append(f"{entry[1]}24")
        return result

    def names(self) -> list[str]:
        return [name for name, _ in self.calls]

    def details(self, name: str) -> list[dict[str, Any]]:
        return [values for called, values in self.calls if called == name]

    def init(self) -> wire.SecretInit:
        packet = self.packets[-1]
        kind, size = wire.parse_header(packet[:HB])
        assert kind is wire.Kind.INIT and size == len(packet) - HB
        return wire.decode_init(packet[HB:])

    def events(self) -> list[dict[str, Any]]:
        return events_t5(self.link.lines)

    def eof_results(self) -> list[dict[str, Any]]:
        return [e for e in self.events() if e["e"] == "RESULT" and e["case"] == T5]

    def aborts(self) -> list[dict[str, Any]]:
        return [e for e in self.events() if e["e"] == "ABORT"]

    def run_t5(self, monkeypatch: pytest.MonkeyPatch) -> Any:
        """The public ``run(ports, owner=...)`` with the matching loaded Owner."""
        module = d1.driver()
        monkeypatch.setitem(module["run"].__globals__, RECORDING_KEY, recording_ledger(self))
        self.owner = module["Owner"]()
        ports = module["Ports"](
            reader=d1.PureReader(self),
            guard=d1.Guard(self),
            clock=self.clock,
            link=self.link,
            ids=T5Ids(self),
            factory=d1.Factory(self),
            pair=d1.Pair(self),
            threads=d1.Threads(self),
            children=ChildPort(self),
        )
        return module["run"](ports, owner=self.owner)

            assert world.t5_releases == []
        else:
            assert world.t5_releases == [(FULL, ReleaseOutcome.RELEASED)]

def test_sanity_fake_child_frames_use_public_codecs() -> None:
    world = T5World()
    port = ChildPort(world)
    parent, child = port.pair()
    reader, writer = port.stdin()
    proc = port.spawn(child, reader, parent_pid=1, address_bytes=WORKER_AS, idle_ms=30000)
    kind, size = wire.parse_header(parent.recv_exact(HB, deadline=2000.0))
    assert kind is wire.Kind.GUARD_READY
    assert wire.decode_guard(parent.recv_exact(size, deadline=2000.0)) == (PROC_PID, 1, WORKER_AS)
    key = wire.SessionKey(T5_ID, d1.DATASOURCE_BYTES, 2, 1)
    snapshot = b"SYNTHETIC-SNAPSHOT" * 3
    limits = ResultLimits(max_rows=1000, max_frame_bytes=16 * MIB)
    packet = wire.encode_init(wire.SecretInit(key, limits, 1012.0, NONCE, snapshot))
    assert writer.write_all(packet, deadline=2000.0) is True
    assert parent.inbox == bytearray()  # OPENED only after the stdin writer closed
    assert writer.close() is True
    kind, size = wire.parse_header(parent.recv_exact(HB, deadline=2000.0))
    assert kind is wire.Kind.OPENED
    tag = wire.decode_opened(parent.recv_exact(size, deadline=2000.0))
    assert hmac.compare_digest(tag, hmac.digest(NONCE, snapshot, "sha256"))
    for name in ("send", "send_all", "write", "shutdown", "shutdown_write", "channel"):
        assert not hasattr(parent, name)
    assert proc.wait(deadline=2000.0) == 0
    assert proc.kill() is True


@pytest.mark.parametrize("mode", ["bad_hmac", "error"])
def test_sanity_fake_negative_opened_frames_are_distinguishable(mode: str) -> None:
    world = T5World(t5={"opened": mode})
    key = wire.SessionKey(T5_ID, d1.DATASOURCE_BYTES, 2, 1)
    limits = ResultLimits(max_rows=1000, max_frame_bytes=16 * MIB)
    world.packets.append(wire.encode_init(wire.SecretInit(key, limits, 1012.0, NONCE, b"SNAPSHOT")))
    frame = world.opened_frame()
    kind, size = wire.parse_header(frame[:HB])
    if mode == "error":
        assert kind is wire.Kind.ERROR
    else:
        assert kind is wire.Kind.OPENED
        assert frame[HB:] != hmac.digest(NONCE, b"SNAPSHOT", "sha256")
        assert len(frame) - HB == size == 32


def test_sanity_v1_delayed_return_trace_and_v3_fence_arithmetic() -> None:
    """Review F1 trace, relative seconds: only the pre-write base keeps below idle end."""
    t_pre, stdin_closed, child_idle_start, write_returned = 0.0, 1.0, 1.01, 11.0
    opened, go, eof, reaped = 11.1, 30.9, 31.1, 31.2
    idle_end = child_idle_start + IDLE_S
    v1_fence = write_returned + IDLE_S - MARGIN_S
    v3_fence = t_pre + IDLE_S - MARGIN_S
    assert t_pre <= stdin_closed < child_idle_start
    assert v1_fence > reaped > idle_end  # v1 would accept an idle exit as EOF
    assert v3_fence < idle_end  # a lower bound of the child idle end
    assert go > min(opened + 20.0, v3_fence - 1.0)  # GO too late for the v3 window
    assert reaped >= v3_fence  # and the reap is idle-late under v3
    assert min(eof + EOF_WAIT_S, v3_fence) == v3_fence

def session_rows(ledger: BudgetLedger) -> list[Any]:
    """Public status only: non-empty SESSION rows (empty slots are None)."""
    rows = [row for row in ledger.status.entries if row is not None]
    return [row for row in rows if row.kind is ReservationKind.SESSION]


def sanity_ledger() -> tuple[BudgetLedger, Any]:
    limits = BudgetLimits(512 * MIB, 128 * MIB, 64 * MIB, 4, 1, 1, 1, 2, 1)
    ledger = BudgetLedger(limits, (DatasourceCapacity(1, 1),))
    generation = ledger.reserve_candidate(0, shared_bytes=MIB)
    ledger.publish(generation)
    token = ledger.reserve_session(generation, worker_bytes=WORKER_AS, parent_bytes=33 * MIB)
    return ledger, token


def test_sanity_real_ledger_drift_refuses_adopt_then_retain_holds() -> None:
    ledger, token = sanity_ledger()
    assert [row.state for row in session_rows(ledger)] == [ReservationState.RESERVED]
    ledger.begin_drain(token)
    assert [row.state for row in session_rows(ledger)] == [ReservationState.DRAINING]
    with pytest.raises(BudgetError) as info:
        ledger.adopt_session(token)
    assert info.value.category is BudgetFailure.STATE
    ledger.retain(token)
    assert [row.state for row in session_rows(ledger)] == [ReservationState.HELD]


def test_sanity_real_output_pin_makes_complete_release_held() -> None:
    ledger, token = sanity_ledger()
    ledger.adopt_session(token)
    assert [row.state for row in session_rows(ledger)] == [ReservationState.LIVE]
    pin = ledger.begin_output(token, frame_bytes=1, overhead_bytes=1)
    ledger.begin_drain(token)
    assert ledger.release(token, FULL) is ReleaseOutcome.HELD
    rows = session_rows(ledger)
    assert [(row.state, row.output_pinned) for row in rows] == [(ReservationState.HELD, True)]
    assert pin is not None  # the real pin stays referenced by this test
