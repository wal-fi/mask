@contextlib.contextmanager
def sealed(lab: SimpleNamespace) -> Iterator[Any]:
    """Blocks effects for the exercise only; restored before pytest reports."""
    with lab.monkeypatch.context() as patch:
        for owner, name in BANS:
            patch.setattr(owner, name, forbidden)
        yield patch


def world(lab: SimpleNamespace, scenario: str = "approved") -> SimpleNamespace:
    bundle = lab.builder["build"](lab.builder[scenario]())
    root = lab.helpers["make_tree"](bundle, lab.verifier)
    fake = lab.helpers["FakeNt"](root)
    return SimpleNamespace(bundle=bundle, root=root, nt=fake, api=fake.api())


def seams(lab: SimpleNamespace, native: Any) -> SimpleNamespace:
    """Replaces every effectful seam; the real source read and argv0 helpers
    must never run in these tests."""
    seen = SimpleNamespace(installed=[], native=[])

    def install(signum: Any, _handler: Any) -> None:
        seen.installed.append(signum)

    def native_api(namespace: Any) -> Any:
        seen.native.append(namespace)
        return native(namespace)

    for name, value in (
        ("_pids", lambda: PIDS),
        ("_install", install),
        ("_native_api", native_api),
        ("_read_sources", forbidden),
        ("_argv0", forbidden),
        ("_drive", forbidden),
    ):
        lab.monkeypatch.setitem(lab.glob, name, value)
    return seen


def returning(api: Any) -> Any:
    def native(_namespace: Any) -> Any:
        return api

    return native


def advance(lab: SimpleNamespace, native: Any) -> tuple[Any, Any, SimpleNamespace]:
    seen = seams(lab, native)
    progress = lab.glob["_progress"](NONCE, ARGV0)
    with sealed(lab):
        decision = lab.glob["_advance"](progress, lab.sources)
    return progress, decision, seen


def events(text: str) -> list[dict[str, Any]]:
    return [json.loads(line) for line in text.splitlines()]


def assert_park(lab: SimpleNamespace, progress: Any, decision: Any, reason: str) -> None:

    with sealed(lab):

    assert lab.glob["_STATE"]["parked"] is True

def functions(tree: ast.Module) -> dict[str, ast.FunctionDef]:
    return {node.name: node for node in tree.body if isinstance(node, ast.FunctionDef)}


def inside(node: ast.AST) -> set[int]:
    return {id(child) for child in ast.walk(node)}


def test_static_structure_is_sticky_and_effect_free() -> None:
    tree = ast.parse(ENTRY.read_bytes())
    imported = {
        alias.name.split(".")[0]
        for node in ast.walk(tree)
        if isinstance(node, ast.Import)
        for alias in node.names
    }
    imported |= {
        (node.module or "").split(".")[0]
        for node in ast.walk(tree)
        if isinstance(node, ast.ImportFrom)
    }
    assert imported <= {"hashlib", "json", "os", "pathlib", "re", "signal", "stat", "sys", "time"}
    names = {node.id for node in ast.walk(tree) if isinstance(node, ast.Name)}
    banned = {"ctypes", "subprocess", "threading", "traceback", "runpy", "importlib", "print"}
    assert not names & banned
    defs = functions(tree)
    native = inside(defs["_native_api"])
    strings = [node for node in ast.walk(tree) if isinstance(node, ast.Constant)]
    assert all(id(node) in native for node in strings if node.value == "_load_native")
    lazy = inside(defs["_argv0"]) | inside(defs["_read_sources"])
    for node in ast.walk(tree):
        if isinstance(node, ast.Attribute) and node.attr == "resolve":
            assert id(node) in lazy
    hold = defs["_hold"]
    assert not any(isinstance(node, (ast.Return, ast.Break)) for node in ast.walk(hold))
    assert any(
        isinstance(node, ast.While) and isinstance(node.test, ast.Constant) and node.test.value
        for node in ast.walk(hold)
    )
    returns = [node for node in ast.walk(defs["_drive"]) if isinstance(node, ast.Return)]
    assert len(returns) == 1
    for node in ast.walk(tree):
        if isinstance(node, ast.Assign) and isinstance(node.targets[0], ast.Subscript):
            target = node.targets[0]
            key = target.slice
            if isinstance(key, ast.Constant) and key.value in ("parked", "announced", "attempted"):
                assert isinstance(node.value, ast.Constant) and node.value.value is True
    calls = [node for node in ast.walk(tree) if isinstance(node, ast.Call)]
    named = [node.func.id for node in calls if isinstance(node.func, ast.Name)]
    assert named.count("_native_api") == 1  # the only call: inside the factory
    mains = [
        node
        for node in tree.body
        if isinstance(node, ast.If) and "__main__" in ast.unparse(node.test)
    ]
    assert len(mains) == 1

def tampered(raw: bytes, other: bytes, kind: str) -> Any:
    variants: dict[str, Any] = {
        "bytearray": bytearray(raw),
        "none": None,
        "short": raw[:-1],
        "flipped": raw[:-1] + bytes([raw[-1] ^ 1]),
        "swapped": other,
    }
    return variants[kind]


@pytest.mark.parametrize("kind", ["bytearray", "none", "short", "flipped", "swapped"])
@pytest.mark.parametrize("index", [0, 1, 2])
def test_every_tampered_source_refuses_before_exec(
    lab: SimpleNamespace, index: int, kind: str, capsys: Any
) -> None:
    sources = list(lab.sources)
    sources[index] = tampered(sources[index], sources[(index + 1) % 3], kind)
    supplied = tuple(sources)
    lab.monkeypatch.setitem(lab.glob, "_read_sources", lambda: supplied)
    for name in ("_argv0", "_drive", "_execute"):
        lab.monkeypatch.setitem(lab.glob, name, forbidden)
    event = refuse_main(lab, [NONCE], capsys)
    assert event["reason"] == "COLLECT_ENTRY_SOURCE" and event["nonce12"] == NONCE[:12]

@pytest.mark.parametrize(("scenario", "code"), [("approved", 0), ("rejected_at_1", 1)])
def test_verified_paths_capture_every_original_before_effects(
    lab: SimpleNamespace, scenario: str, code: int, capsys: Any
) -> None:
    w = world(lab, scenario)
    progress, decision, seen = advance(lab, returning(w.api))
    summary, transport = sha(w.bundle.summary_raw), sha(w.bundle.transport_raw)
    assert decision == ("EXIT", code, "OK", summary, transport)
    keep = lab.glob["KEEP"]
    namespace = progress["namespace"]
    assert any(item is namespace for item in keep) and any(item is progress for item in keep)
    attempts = progress["attempts"]
    assert namespace["_ATTEMPTS"] is attempts and type(attempts) is list
    record = progress["record"]
    assert attempts == [record] and attempts[0] is record
    assert progress["collector"] is record.collector
    assert progress["active"] is record.collector["_ACTIVE"]
    assert progress["poison"] is record.collector["_POISON"]
    backend = progress["backend"]
    assert backend is record.backend
    assert progress["open"] is backend._open and progress["retained"] is backend._retained
    assert seen.native == [record.native]
    assert seen.installed == [signal.SIGINT, signal.SIGBREAK]
    assert lab.glob["_STATE"]["active"] is False and lab.glob["_STATE"]["attempted"] is True
    with sealed(lab):
        assert lab.glob["settle"](decision, progress) == decision
        lab.glob["_final"](progress, decision)
    captured = capsys.readouterr()
    identity, final = events(captured.out)
    assert set(identity) == IDENTITY_KEYS and identity["event"] == "IDENTITY"
    assert (identity["controller_pid"], identity["parent_pid"]) == PIDS
    assert set(final) == FINAL_KEYS and final["code"] == code
    assert final["outcome"] == EXIT_OUTCOMES[code] and final["reason"] == "OK"
    assert (final["summary_sha256"], final["transport_sha256"]) == (summary, transport)
    assert ARGV0 not in captured.out and "Users" not in captured.out and captured.err == ""

            progress = holder[0]

    api = SimpleNamespace(**{name: wrap(name) for name in lab.helpers["API_OPERATIONS"]})
    seen = seams(lab, returning(api))

    assert decision[:3] == ("EXIT", 0, "OK") and len(seen.native) == 1
    assert states and all(captured for captured, _size in states)
    assert states[0][1] >= 1  # the first pending call was already retained, kept as is

def replace(progress: Any, change: str) -> None:
    record, backend = progress["record"], progress["backend"]
    if change == "active":
        record.collector["_ACTIVE"] = []
    elif change == "poison":
        record.collector["_POISON"] = []
    elif change == "open":
        backend._open = {}
    elif change == "retained":
        backend._retained = []
    elif change == "open-subclass":
        backend._open = SubDict()
        progress["open"] = backend._open
    elif change == "retained-subclass":
        backend._retained = SubList()
        progress["retained"] = backend._retained
    elif change == "collector":
        record.collector = dict(record.collector)
    elif change == "attempts":
        progress["namespace"]["_ATTEMPTS"] = [record]
    elif change == "extra-record":
        progress["attempts"].append(record)
    else:
        record.backend = None


CHANGES = [
    "active",
    "poison",
    "open",
    "retained",
    "open-subclass",
    "retained-subclass",
    "collector",
    "attempts",
    "extra-record",
    "backend",
]


@pytest.mark.parametrize("change", CHANGES)
def test_replaced_or_hostile_originals_park(lab: SimpleNamespace, change: str) -> None:

        assert lab.glob["settle"](decision, progress) == ("PARK", "INTERNAL")
    assert lab.glob["_STATE"]["parked"] is True

    def __init__(self, error: BaseException) -> None:

def once_raising(original: Any, error: BaseException, fired: list[bool]) -> Any:
    """A property that raises `error` on its first read, then returns the
    held original object."""

    def read(_self: Any) -> Any:
        if not fired:
            fired.append(True)
            raise error
        return original

    return property(read)


def test_f1_a_control_while_settling_parks_and_keeps_the_first_control(
    lab: SimpleNamespace,
) -> None:
    w = world(lab)

    assert decision[:3] == ("EXIT", 0, "OK")
    first, fired = Signal(), []
    reader = once_raising(progress["open"], first, fired)
    lab.monkeypatch.setattr(type(progress["backend"]), "_open", reader, raising=False)
    with sealed(lab):
        assert lab.glob["settle"](decision, progress) == ("PARK", "INTERNAL")
        assert progress["control"] is first
        assert lab.glob["settle"](decision, progress) == ("PARK", "INTERNAL")
    assert fired == [True]
    assert lab.glob["_STATE"]["parked"] is True
    assert progress["control"] is first

def forged(summary: str, transport: str, case: str) -> Any:
    table: dict[str, Any] = {
        "exit-subclass": (SneakyStr("EXIT"), 0, "OK", summary, transport),
        "bool-code": ("EXIT", True, "OK", summary, transport),
        "code0-reason": ("EXIT", 0, "NOT_OK", "not-a-sha", "not-a-sha"),
        "code2-unknown": ("EXIT", 2, "NOT_A_REASON", None, None),
        "code2-sentinel": ("EXIT", 2, SENTINEL, None, None),
        "code2-hash": ("EXIT", 2, "COLLECT_OPEN", summary, None),
        "code5-reason": ("EXIT", 5, "OK", None, None),
        "code5-hash": ("EXIT", 5, "INTERRUPTED", None, transport),
        "stale-rejected": ("EXIT", 1, "OK", summary, transport),
        "stale-refused": ("EXIT", 2, "COLLECT_OPEN", None, None),
        "stale-interrupted": ("EXIT", 5, "INTERRUPTED", None, None),
    }
    bad = {
        "upper": summary.upper(),
        "short": summary[:-1],
        "nonhex": "g" * 64,
        "bytes": summary.encode(),
        "none": None,
    }
    for kind, value in bad.items():
        table["summary-" + kind] = ("EXIT", 0, "OK", value, transport)
        table["transport-" + kind] = ("EXIT", 0, "OK", summary, value)
    return table[case]

def swap(progress: Any, change: str) -> None:
    record, backend = progress["record"], progress["backend"]
    if change == "active":
        record.collector["_ACTIVE"] = []
    elif change == "attempts":
        progress["namespace"]["_ATTEMPTS"] = [record]
    elif change == "native":
        record.native = dict(record.native)
    elif change == "open":
        backend._open = {}
    else:
        backend._retained = []


@pytest.mark.parametrize("change", ["active", "attempts", "native", "open", "retained"])
def test_f3_an_unknown_identity_chain_refuses_every_later_delegation(
    lab: SimpleNamespace, change: str
) -> None:
    w = world(lab)
    holder: list[Any] = []
    anchor = lab.helpers["ANCHOR"]

    def swapping(default: Any, _extra: Any) -> Any:
        status = default()
        swap(holder[0], change)  # after the first delegated effect
        return status

    w.nt.hooks[("create", anchor, 1)] = swapping
    seams(lab, returning(w.api))
    progress = lab.glob["_progress"](NONCE, ARGV0)
    holder.append(progress)
    with sealed(lab):
        decision = lab.glob["_advance"](progress, lab.sources)
    assert w.nt.calls == [("create", anchor)]  # nothing delegated after it, no close
    assert w.nt.closed == [] and len(w.nt.handles) == 1
    assert progress["backend"] is not None and progress["backend_captured"] is True
    assert_park(lab, progress, decision, "UNKNOWN_CUSTODY")
